76pub struct ThrustCompiler<'thrustc> {
77    /// Archivos objeto listos para ser enlazados, en orden de compilación.
78    linked_objects: Vec<std::path::PathBuf>,
79    unready: &'thrustc [CompilationUnit],
80
81    /// Ruta del archivo fuente -> ruta del archivo objeto generado.
82    /// Índice de deduplicación: cuando se reprocesa una unidad (p. ej., un módulo
83    /// recompilado tras quedar pendientes instanciaciones genéricas), su archivo
84    /// objeto anterior se reemplaza para que el enlazador no encuentre símbolos duplicados.
85    source_to_object: std::collections::HashMap<std::path::PathBuf, std::path::PathBuf>,
86    linked_sanitizers: Vec<Sanitizer>,
87    file_output_requested: bool,
88
89    options: &'thrustc CompilerOptions,
90
91    linking_time: std::time::Duration,
92    thrustc_frontend_time: std::time::Duration,
93    thrustc_backend_time: std::time::Duration,
94    thrustc_time: std::time::Duration,
95}

97pub type CompileTime = (
98    bool,
99    std::time::Duration,
100    std::time::Duration,
101    std::time::Duration,
102    std::time::Duration,
103);

105impl<'thrustc> ThrustCompiler<'thrustc> {
106    pub fn new(units: &'thrustc [CompilationUnit], options: &'thrustc CompilerOptions) -> Self {
107        Self {
108            linked_objects: Vec::with_capacity(units.len()),
109            unready: units,
110
111            source_to_object: std::collections::HashMap::with_capacity(units.len()),
112            linked_sanitizers: Vec::with_capacity(5),
113            file_output_requested: false,
114
115            options,
116
117            linking_time: std::time::Duration::default(),
118            thrustc_frontend_time: std::time::Duration::default(),
119            thrustc_backend_time: std::time::Duration::default(),
120            thrustc_time: std::time::Duration::default(),
121        }
122    }
123}

125impl ThrustCompiler<'_> {
126    pub fn compile(&mut self) -> CompileTime {
127        thrustc_generics::reset_pending_instantiations();
128
129        if self.unready.is_empty() {
130            thrustc_logging::print_critical_error(
131                thrustc_logging::LoggingType::Error,
132                "No input files",
133            );
134        }
135
136        if self.get_compilation_options().llvm() {
137            Target::initialize_all(&InitializationConfig::default());
138
139            if self
140                .get_compilation_options()
141                .get_llvm_backend()
142                .is_execution_in_jit()
143            {
144                return self.compile_jit_llvm();
145            } else {
146                return self.compile_aot_llvm();
147            }
148        } else {
149            thrustc_logging::print_warning(
150                thrustc_logging::LoggingType::Warning,
151                "Unrecognizable code generator selection. You should select either LLVM or GCC.",
152            );
153        }
154
155        (
156            false,
157            self.thrustc_time,
158            self.thrustc_frontend_time,
159            self.thrustc_backend_time,
160            self.linking_time,
161        )
162    }
163}

165impl<'thrustc> ThrustCompiler<'thrustc> {
166    fn compile_aot_llvm(&mut self) -> CompileTime {
167        cleaner::auto_clean(self.get_compilation_options());
168
169        self.discover_std();
170
171        if self.compile_imported_std().is_err() {
172            return (
173                true,
174                self.thrustc_time,
175                self.thrustc_frontend_time,
176                self.thrustc_backend_time,
177                self.linking_time,
178            );
179        }
180
181        let mut it_failed: bool = false;
182
183        for file in self.unready.iter() {
184            it_failed |= self.compile_file_with_llvm_aot(file).is_err();
185        }
186
187        it_failed = it_failed || self.reprocess_pending_instantiations().is_err();
188
189        let output_requested: bool = self.get_compilation_options().was_printed()
190            || self.get_compilation_options().was_emited()
191            || self.file_output_requested;
192
193        it_failed = it_failed || (self.get_compiled_files().is_empty() && !output_requested);
194
195        if it_failed {
196            return (
197                it_failed,
198                self.thrustc_time,
199                self.thrustc_frontend_time,
200                self.thrustc_backend_time,
201                self.linking_time,
202            );
203        }
204
205        if output_requested {
206            return (
207                false,
208                self.thrustc_time,
209                self.thrustc_frontend_time,
210                self.thrustc_backend_time,
211                self.linking_time,
212            );
213        }
214
215        starter::linking_phase(self.get_compilation_options(), self.get_compiled_files());
216
217        let linking_compiler_config: &LinkingCompilersConfiguration =
218            self.options.get_linking_compilers_configuration();
219
220        if linking_compiler_config.get_use_clang() {
221            it_failed = linkage::link_with_clang(self).is_err();
222        } else if linking_compiler_config.get_use_gcc() {
223            it_failed = linkage::link_with_gcc(self).is_err();
224        } else {
225            thrustc_logging::print_warning(
226                thrustc_logging::LoggingType::Warning,
227                "Unknown linking method, ignoring linking.",
228            );
229        }
230
231        (
232            it_failed,
233            self.thrustc_time,
234            self.thrustc_frontend_time,
235            self.thrustc_backend_time,
236            self.linking_time,
237        )
238    }

240    fn discover_std(&mut self) {
241        for file in self.unready.iter() {
242            let Ok(tokens) = Lexer::lex(file, self.options) else {
243                continue;
244            };
245
246            let Ok(directives) = thrustc_directive::apply_file_directives(&tokens) else {
247                continue;
248            };
249            let file_options: FileOptions<'_, '_> = FileOptions::new(self.options, &directives);
250
251            let mut preprocessor: Preprocessor = Preprocessor::new();
252
253            let builtins: BuiltinRegistry = self.create_builtins_registry();
254
255            let _ = preprocessor.generate_modules(&tokens, &file_options, file, &builtins);
256        }
257    }
258
259    fn compile_imported_std(&mut self) -> Result<(), ()> {
260        if !thrustc_preprocessor::std_library::has_imported_std() {
261            return Ok(());
262        }
263
264        let mut compiled_paths: std::collections::HashSet<std::path::PathBuf> =
265            std::collections::HashSet::new();
266
267        loop {
268            let mut compiled_any: bool = false;
269
270            for module in thrustc_preprocessor::std_library::get_imported_std_modules() {
271                let path: std::path::PathBuf = module.get_path().to_path_buf();
272
273                if !path.is_file() || !compiled_paths.insert(path.clone()) {
274                    continue;
275                }
276
277                self.compile_imported_std_module(&path)?;
278                compiled_any = true;
279            }
280
281            if !compiled_any {
282                break;
283            }
284        }
285
286        Ok(())
287    }

289    fn compile_imported_std_module(&mut self, path: &std::path::Path) -> Result<(), ()> {
290        let name: String = path
291            .file_name()
292            .map_or_else(String::new, |name| name.to_string_lossy().to_string());
293
294        let base_name: String = path.file_stem().map_or_else(String::new, |base_name| {
295            base_name.to_string_lossy().to_string()
296        });
297
298        let content: String = thrustc_reader::get_file_source_code(path);
299        let unit: CompilationUnit =
300            CompilationUnit::new(name, path.to_path_buf(), content, base_name);
301
302        self.compile_file_with_llvm_aot(&unit)
303    }
304
305    fn compile_imported_std_jit<'a>(
306        &mut self,
307        context: &'a Context,
308    ) -> Result<Vec<(std::path::PathBuf, Module<'a>)>, ()> {
309        if !thrustc_preprocessor::std_library::has_imported_std() {
310            return Ok(Vec::with_capacity(0));
311        }
312
313        let mut modules: Vec<(std::path::PathBuf, Module<'a>)> =
314            Vec::with_capacity(u8::MAX as usize);
315
316        let mut compiled_paths: std::collections::HashSet<std::path::PathBuf> =
317            std::collections::HashSet::new();
318
319        loop {
320            let mut compiled_any: bool = false;
321
322            for module in thrustc_preprocessor::std_library::get_imported_std_modules() {
323                let path: std::path::PathBuf = module.get_path().to_path_buf();
324
325                if !path.is_file() {
326                    continue;
327                }
328
329                if !compiled_paths.insert(path.clone()) {
330                    continue;
331                }
332
333                let name: String = path
334                    .file_name()
335                    .map_or_else(String::new, |name| name.to_string_lossy().to_string());
336
337                let base_name: String = path.file_stem().map_or_else(String::new, |base_name| {
338                    base_name.to_string_lossy().to_string()
339                });
340
341                let content: String = thrustc_reader::get_file_source_code(&path);
342                let unit: CompilationUnit =
343                    CompilationUnit::new(name, path.clone(), content, base_name);
344
345                let compiled_file: either::Either<MemoryBuffer, ()> =
346                    self.compile_file_with_llvm_jit(&unit)?;
347
348                if let Some(module) = compiled_file
349                    .left()
350                    .and_then(|memory_buffer| context.create_module_from_ir(memory_buffer).ok())
351                {
352                    let module_path: std::path::PathBuf =
353                        path.canonicalize().unwrap_or(path.clone());
354
355                    modules.push((module_path, module));
356                }
357
358                compiled_any = true;
359            }
360
361            if !compiled_any {
362                break;
363            }
364        }
365
366        Ok(modules)
367    }

369    fn compile_file_with_llvm_aot(&mut self, file: &CompilationUnit) -> Result<(), ()> {
370        let file_time: std::time::Instant = std::time::Instant::now();
371        let frontend_time: std::time::Instant = std::time::Instant::now();
372
373        starter::archive_compilation_unit(self.get_compilation_options(), file);
374
375        let llvm_backend: &LLVMBackend = self.options.get_llvm_backend();
376        let build_dir: &std::path::PathBuf = self.options.get_build_dir();
377
378        let Ok(tokens) = Lexer::lex(file, self.options) else {
379            return interrupt::archive_compilation_module(self, file, file_time);
380        };
381
382        let directives: FileDirectives = match thrustc_directive::apply_file_directives(&tokens) {
383            Ok(directives) => directives,
384            Err(error) => {
385                let mut diagnostician: Diagnostician = Diagnostician::new(file, self.options);
386
387                diagnostician.dispatch_diagnostic(&error, thrustc_logging::LoggingType::Error);
388
389                return interrupt::archive_compilation_module(self, file, file_time);
390            }
391        };
392
393        let file_options: FileOptions<'_, '_> = FileOptions::new(self.options, &directives);
394
395        self.register_linked_sanitizer(&file_options);
396
397        self.file_output_requested |=
398            !directives.emit().is_empty() || !directives.print().is_empty();
399
400        self.update_thrustc_frontend_time(frontend_time.elapsed());
401
402        if print::before_frontend(self, &file_options, file, Emited::Tokens(&tokens))? {
403            return finisher::archive_compilation(self, file_time, file);
404        }
405
406        if emit::before_frontend(
407            self,
408            &file_options,
409            build_dir,
410            file,
411            Emited::Tokens(&tokens),
412        )? {
413            return finisher::archive_compilation(self, file_time, file);
414        }
415
416        if file_options.stop_compilation_at(CompilationPhase::Lexer) {
417            return finisher::archive_compilation(self, file_time, file);
418        }
419
420        let mut builtins: BuiltinRegistry = self.create_builtins_registry();
421
422        let mut preprocessor: Preprocessor = Preprocessor::new();
423
424        let modules: Result<&[thrustc_preprocessor::module::Module], ()> =
425            preprocessor.generate_modules(&tokens, &file_options, file, &builtins);
426
427        self.update_thrustc_frontend_time(frontend_time.elapsed());
428
429        if modules.is_err() {
430            return interrupt::archive_compilation_module(self, file, file_time);
431        }
432
433        let modules: &[thrustc_preprocessor::module::Module] = modules.map_err(|_| {
434            let _ = interrupt::archive_compilation_unit_with_message(
435                self,
436                thrustc_logging::LoggingType::Error,
437                "Failed to get all modules from the preprocessor. Maybe this is a issue.",
438                file,
439                file_time,
440            );
441        })?;
442
443        let parser: (ParserContext, bool) = Parser::parse(
444            &tokens,
445            modules,
446            file,
447            self.options,
448            &file_options,
449            &mut builtins,
450        );
451
452        let parser_result: (ParserContext, bool) = parser;
453        let parser_failed: bool = parser_result.1;
454
455        let parser_context: ParserContext = parser_result.0;
456
457        let ast: &[Ast] = parser_context.get_ast();
458
459        if emit::before_frontend(self, &file_options, build_dir, file, Emited::Ast(ast))? {
460            return finisher::archive_compilation(self, file_time, file);
461        }
462
463        if print::before_frontend(self, &file_options, file, Emited::Ast(ast))? {
464            return finisher::archive_compilation(self, file_time, file);
465        }
466
467        if file_options.stop_compilation_at(CompilationPhase::Parser) {
468            return finisher::archive_compilation(self, file_time, file);
469        }
470
471        {
472            let mut semantic_analysis: SemanticAnalysis<'_> =
473                SemanticAnalysis::new(ast, file, &file_options);
474
475            let semantic_analysis_failed: either::Either<bool, ()> =
476                semantic_analysis.execute(parser_failed);
477
478            self.update_thrustc_frontend_time(frontend_time.elapsed());
479
480            match semantic_analysis_failed {
481                either::Either::Left(semantic_analysis_failed) => {
482                    if parser_failed || semantic_analysis_failed {
483                        return interrupt::archive_compilation_module(self, file, file_time);
484                    }
485                }
486                either::Either::Right(()) => {
487                    return finisher::archive_compilation(self, file_time, file);
488                }
489            }
490        }
491
492        {
493            let mut intrinsic_checker: LLVMIntrinsicChecker<'_> =
494                LLVMIntrinsicChecker::new(ast, file, self.options);
495
496            let mut call_conv_checker: LLVMCallConventionsChecker<'_> =
497                LLVMCallConventionsChecker::new(ast, self.options, file);
498
499            let intrinsic_checker_fail: bool = intrinsic_checker.analyze();
500
501            self.update_thrustc_frontend_time(frontend_time.elapsed());
502
503            if file_options.stop_compilation_at(CompilationPhase::LLVMIntrinsicChecker) {
504                return finisher::archive_compilation(self, file_time, file);
505            }
506
507            let call_convention_checker_fail: bool = call_conv_checker.analyze();
508
509            self.update_thrustc_frontend_time(frontend_time.elapsed());
510
511            if file_options.stop_compilation_at(CompilationPhase::LLVMCallConventionChecker) {
512                return finisher::archive_compilation(self, file_time, file);
513            }
514
515            if intrinsic_checker_fail || call_convention_checker_fail {
516                return interrupt::archive_compilation_module(self, file, file_time);
517            }
518        }
519
520        if print::after_frontend(self, &file_options, file, Emited::Ast(ast))? {
521            return finisher::archive_compilation(self, file_time, file);
522        }
523
524        if emit::after_frontend(self, &file_options, build_dir, file, Emited::Ast(ast))? {
525            return finisher::archive_compilation(self, file_time, file);
526        }
527
528        let backend_time: std::time::Instant = std::time::Instant::now();
529
530        let target: &LLVMTarget = llvm_backend.get_target();
531        let llvm_triple: &TargetTriple = target.get_target_triple();
532
533        let llvm_target_triple_formatted: String =
534            llvm_triple.as_str().to_string_lossy().to_string();
535
536        let llvm_target_triple: LLVMTargetTriple =
537            LLVMTargetTriple::new(llvm_target_triple_formatted.clone());
538
539        let (llvm_cpu_name, llvm_cpu_features): (&str, &str) =
540            llvm_backend.get_cross_target_cpu();
541
542        let compiler_optimization: ThrustOptimization = file_options.optimization();
543        let llvm_opt: OptimizationLevel = compiler_optimization.to_llvm_opt();
544
545        let target: Target = Target::from_triple(llvm_triple).map_err(|_| {
546            let _ = interrupt::archive_compilation_unit_with_message(
547                self,
548                thrustc_logging::LoggingType::Error,
549                "The compiler couldn't be configured correctly. The target is possibly unrecognizable. Try again another target or try to fix it.",
550                file,
551                file_time,
552            );
553        })?;
554
555        if !target.has_target_machine() {
556            interrupt::archive_compilation_unit_with_message(
557                self,
558                thrustc_logging::LoggingType::Error,
559                "The compiler couldn't be configured correctly. The specified target cannot be used for code generation. Try with another target.",
560                file,
561                file_time,
562            )?;
563        }
564
565        let target_machine: TargetMachine = target
566            .create_target_machine(
567                llvm_triple,
568                llvm_cpu_name,
569                llvm_cpu_features,
570                llvm_opt,
571                file_options.reloc_model(),
572                file_options.code_model(),
573            )
574            .ok_or_else(|| {
575                let _ = interrupt::archive_compilation_unit_with_message(
576                    self,
577                    thrustc_logging::LoggingType::Error,
578                    "The compiler couldn't be configured correctly. Possibly the target is not supported for code generation.",
579                    file,
580                    file_time,
581                );
582            })?;
583
584        let target_data: TargetData = target_machine.get_target_data();
585        let target_triple: TargetTriple = target_machine.get_triple();
586
587        let target_info: TargetInfo =
588            TargetInfo::new(LLVMTargetTriple::new(llvm_target_triple_formatted));
589
590        let mut target_abi: Option<LLVMABIRepresentation> = thrustc_llvm_abi::get_abi(
591            self.options.abi_configuration().specific(),
592            file,
593            self.options,
594            &llvm_target_triple,
595            &target_info,
596            &target_data,
597        );
598
599        if self.options.abi_configuration().disable() {
600            target_abi = None;
601        }
602
603        let llvm_context: Context = Context::create();
604        let llvm_builder: Builder = llvm_context.create_builder();
605        let llvm_module: Module = llvm_context.create_module(file.get_name());
606
607        llvm_module.set_triple(llvm_triple);
608        llvm_module.set_data_layout(&target_machine.get_target_data().get_data_layout());
609
610        let mut llvm_codegen_context: LLVMCodeGenContext = LLVMCodeGenContext::new(
611            &llvm_module,
612            &llvm_context,
613            &llvm_builder,
614            &target_data,
615            &target_triple,
616            &target_machine,
617            target_abi.as_ref(),
618            Diagnostician::new(file, self.options),
619            self.options,
620            &file_options,
621            file,
622        );
623
624        thrustc_llvm_codegen::LLVMCompiler::compile(&mut llvm_codegen_context, ast);
625
626        validate::llvm_codegen(&llvm_module, file)?;
627
628        self.update_thrustc_backend_time(backend_time.elapsed());
629
630        if print::llvm_before_optimization(
631            self,
632            &file_options,
633            &llvm_module,
634            &target_machine,
635            file,
636            file_time,
637        )? {
638            return finisher::archive_compilation(self, file_time, file);
639        }
640
641        if emit::llvm_before_optimization(
642            self,
643            &file_options,
644            &llvm_module,
645            &target_machine,
646            build_dir,
647            file,
648            file_time,
649        )? {
650            return finisher::archive_compilation(self, file_time, file);
651        }
652
653        if file_options.stop_compilation_at(CompilationPhase::LLVMCodegen) {
654            return finisher::archive_compilation(self, file_time, file);
655        }
656
657        self.update_thrustc_backend_time(backend_time.elapsed());
658
659        let llvm_optimizer_config: LLVMOptimizationConfig = LLVMOptimizationConfig::new(
660            compiler_optimization,
661            file_options.sanitizer(),
662            file_options.symbol_linkage_strategy(),
663            file_options.denormal_fp(),
664            file_options.denormal_fp_32(),
665        );
666
667        let llvm_optimizer_passes: LLVMOptimizerPasses<'_> = LLVMOptimizerPasses::new(
668            file_options.opt_passes(),
669            file_options.modificator_opt_passes(),
670        );
671
672        let llvm_optimizer_flags: LLVMOptimizerFlags = LLVMOptimizerFlags::new(
673            file_options.omit_default_optimizations(),
674            file_options.disable_all_sanitizers(),
675        );
676
677        thrustc_llvm_codegen::optimizer::LLVMOptimizer::new(
678            &llvm_module,
679            &llvm_context,
680            &target_machine,
681            llvm_optimizer_config,
682            llvm_optimizer_flags,
683            llvm_optimizer_passes,
684        )
685        .optimize();
686
687        self.update_thrustc_backend_time(backend_time.elapsed());
688
689        if print::llvm_after_optimization(
690            self,
691            &file_options,
692            &llvm_module,
693            &target_machine,
694            file,
695            file_time,
696        )? {
697            return finisher::archive_compilation(self, file_time, file);
698        }
699
700        if emit::llvm_after_optimization(
701            self,
702            &file_options,
703            &llvm_module,
704            &target_machine,
705            build_dir,
706            file,
707            file_time,
708        )? {
709            return finisher::archive_compilation(self, file_time, file);
710        }
711
712        let obj_file: std::path::PathBuf = finisher::llvm_obj_compilation(
713            self.options,
714            &llvm_module,
715            &target_machine,
716            build_dir,
717            file.get_name(),
718        );
719
720        self.register_compiled_unit(file.get_path().to_path_buf(), obj_file);
721
722        self.update_thrustc_backend_time(backend_time.elapsed());
723
724        finisher::archive_compilation(self, file_time, file)?;
725
726        Ok(())
727    }
728}

730impl<'thrustc> ThrustCompiler<'thrustc> {
731    fn compile_jit_llvm(&mut self) -> CompileTime {
732        cleaner::auto_clean(self.get_compilation_options());
733
734        self.discover_std();
735
736        let context: Context = Context::create();
737
738        let mut it_failed: bool = false;
739        let mut modules: Vec<(std::path::PathBuf, Module)> = Vec::with_capacity(u8::MAX as usize);
740
741        let std_modules: Result<Vec<(std::path::PathBuf, Module)>, ()> =
742            self.compile_imported_std_jit(&context);
743
744        let std_modules: Vec<(std::path::PathBuf, Module)> = match std_modules {
745            Ok(std_modules) => std_modules,
746            Err(()) => {
747                return (
748                    true,
749                    self.thrustc_time,
750                    self.thrustc_frontend_time,
751                    self.thrustc_backend_time,
752                    self.linking_time,
753                );
754            }
755        };
756
757        modules.extend(std_modules);
758
759        for file in self.unready.iter() {
760            let compiled_file: Result<either::Either<MemoryBuffer, ()>, ()> =
761                self.compile_file_with_llvm_jit(file);
762
763            it_failed |= compiled_file.is_err();
764
765            if let Some(module) = compiled_file
766                .ok()
767                .and_then(|either| either.left())
768                .and_then(|memory_buffer| context.create_module_from_ir(memory_buffer).ok())
769            {
770                let path: std::path::PathBuf = file
771                    .get_path()
772                    .canonicalize()
773                    .unwrap_or_else(|_| file.get_path().to_path_buf());
774
775                if let Some(index) = modules.iter().position(|(candidate, _)| candidate == &path) {
776                    modules[index] = (path, module);
777                } else {
778                    modules.push((path, module));
779                }
780            }
781        }
782
783        let mut reprocesses: usize = 0;
784
785        while thrustc_generics::has_pending_instantiations() && reprocesses < 1024 {
786            let paths: Vec<std::path::PathBuf> = thrustc_generics::pending_module_paths();
787
788            for path in paths {
789                let name: String = path
790                    .file_name()
791                    .map_or_else(String::new, |name| name.to_string_lossy().to_string());
792
793                let base_name: String = path.file_stem().map_or_else(String::new, |base_name| {
794                    base_name.to_string_lossy().to_string()
795                });
796
797                let content: String = thrustc_reader::get_file_source_code(&path);
798                let unit: CompilationUnit =
799                    CompilationUnit::new(name, path.clone(), content, base_name);
800
801                let compiled_file: either::Either<MemoryBuffer, ()> =
802                    match self.compile_file_with_llvm_jit(&unit) {
803                        Ok(compiled_file) => compiled_file,
804                        Err(()) => {
805                            return (
806                                true,
807                                self.thrustc_time,
808                                self.thrustc_frontend_time,
809                                self.thrustc_backend_time,
810                                self.linking_time,
811                            );
812                        }
813                    };
814
815                let Some(module) = compiled_file
816                    .left()
817                    .and_then(|memory_buffer| context.create_module_from_ir(memory_buffer).ok())
818                else {
819                    return (
820                        true,
821                        self.thrustc_time,
822                        self.thrustc_frontend_time,
823                        self.thrustc_backend_time,
824                        self.linking_time,
825                    );
826                };
827
828                let module_path: std::path::PathBuf = path.canonicalize().unwrap_or(path.clone());
829
830                if let Some(index) = modules
831                    .iter()
832                    .position(|(candidate, _)| candidate == &module_path)
833                {
834                    modules[index] = (module_path, module);
835                } else {
836                    modules.push((module_path, module));
837                }
838            }
839
840            reprocesses = reprocesses.saturating_add(1);
841        }
842
843        if thrustc_generics::has_pending_instantiations() {
844            it_failed = true;
845        }
846
847        let output_requested: bool = self.get_compilation_options().was_printed()
848            || self.get_compilation_options().was_emited()
849            || self.file_output_requested;
850
851        it_failed = it_failed || (modules.is_empty() && !output_requested);
852
853        if it_failed {
854            return (
855                it_failed,
856                self.thrustc_time,
857                self.thrustc_frontend_time,
858                self.thrustc_backend_time,
859                self.linking_time,
860            );
861        }
862
863        if output_requested {
864            return (
865                false,
866                self.thrustc_time,
867                self.thrustc_frontend_time,
868                self.thrustc_backend_time,
869                self.linking_time,
870            );
871        }
872
873        let mut modules: Vec<Module> = modules.into_iter().map(|(_, module)| module).collect();
874
875        modules.reverse();
876
877        let Some(module) = modules.first() else {
878            thrustc_logging::print_warning(
879                thrustc_logging::LoggingType::Warning,
880                "There's nothing to compile for the JIT compiler. Skipping compilation.",
881            );
882
883            return (
884                it_failed,
885                self.thrustc_time,
886                self.thrustc_frontend_time,
887                self.thrustc_backend_time,
888                self.linking_time,
889            );
890        };
891
892        let llvm_backend: &LLVMBackend = self.options.get_llvm_backend();
893        let config: &JITConfiguration = llvm_backend.get_jit_config();
894        let opt_level: OptimizationLevel = llvm_backend.get_optimization().to_llvm_opt();
895
896        let engine: ExecutionEngine = match module.create_jit_execution_engine(opt_level) {
897            Ok(engine) => engine,
898            Err(_) => {
899                thrustc_logging::print_error(
900                    thrustc_logging::LoggingType::Error,
901                    "The JIT compiler couldn't be created correctly. Unexpected issue.",
902                );
903
904                return (
905                    it_failed,
906                    self.thrustc_time,
907                    self.thrustc_frontend_time,
908                    self.thrustc_backend_time,
909                    self.linking_time,
910                );
911            }
912        };
913
914        let llvm_jit: LLVMJITCompiler =
915            thrustc_llvm_codegen::jit::LLVMJITCompiler::new(engine, config, modules);
916
917        let llvm_jit_result: i32 = llvm_jit.compile_and_run().unwrap_or(1);
918
919        std::process::exit(llvm_jit_result)
920    }
921
922    fn compile_file_with_llvm_jit(
923        &mut self,
924        file: &CompilationUnit,
925    ) -> Result<either::Either<MemoryBuffer, ()>, ()> {
926        let file_time: std::time::Instant = std::time::Instant::now();
927        let frontend_time: std::time::Instant = std::time::Instant::now();
928
929        starter::archive_compilation_unit(self.get_compilation_options(), file);
930
931        let llvm_backend: &LLVMBackend = self.options.get_llvm_backend();
932        let build_dir: &std::path::PathBuf = self.options.get_build_dir();
933
934        let Ok(tokens) = Lexer::lex(file, self.options) else {
935            return interrupt::archive_compilation_module_jit(self, file, file_time);
936        };
937
938        let directives: FileDirectives = match thrustc_directive::apply_file_directives(&tokens) {
939            Ok(directives) => directives,
940            Err(error) => {
941                let mut diagnostician: Diagnostician = Diagnostician::new(file, self.options);
942
943                diagnostician.dispatch_diagnostic(&error, thrustc_logging::LoggingType::Error);
944
945                return interrupt::archive_compilation_module_jit(self, file, file_time);
946            }
947        };
948
949        let file_options: FileOptions<'_, '_> = FileOptions::new(self.options, &directives);
950
951        self.file_output_requested |=
952            !directives.emit().is_empty() || !directives.print().is_empty();
953
954        self.update_thrustc_frontend_time(frontend_time.elapsed());
955
956        if print::before_frontend(self, &file_options, file, Emited::Tokens(&tokens))? {
957            return finisher::archive_compilation_module_jit(self, file_time, file);
958        }
959
960        if emit::before_frontend(
961            self,
962            &file_options,
963            build_dir,
964            file,
965            Emited::Tokens(&tokens),
966        )? {
967            return finisher::archive_compilation_module_jit(self, file_time, file);
968        }
969
970        if file_options.stop_compilation_at(CompilationPhase::Lexer) {
971            return finisher::archive_compilation_module_jit(self, file_time, file);
972        }
973
974        let mut builtins: BuiltinRegistry = self.create_builtins_registry();
975
976        let mut preprocessor: Preprocessor = Preprocessor::new();
977        let modules: Result<&[thrustc_preprocessor::module::Module], ()> =
978            preprocessor.generate_modules(&tokens, &file_options, file, &builtins);
979
980        self.update_thrustc_frontend_time(frontend_time.elapsed());
981
982        if modules.is_err() {
983            return interrupt::archive_compilation_module_jit(self, file, file_time);
984        }
985
986        let modules: &[thrustc_preprocessor::module::Module] = modules.map_err(|_| {
987            let _ = interrupt::archive_compilation_unit_with_message(
988                self,
989                thrustc_logging::LoggingType::Error,
990                "Failed to get all modules from the preprocessor. Maybe this is a issue.",
991                file,
992                file_time,
993            );
994        })?;
995
996        let parser: (ParserContext, bool) = Parser::parse(
997            &tokens,
998            modules,
999            file,
1000            self.options,
1001            &file_options,
1002            &mut builtins,
1003        );
1004
1005        let parser_result: (ParserContext, bool) = parser;
1006        let parser_failed: bool = parser_result.1;
1007
1008        let parser_context: ParserContext = parser_result.0;
1009
1010        let ast: &[Ast] = parser_context.get_ast();
1011
1012        if print::before_frontend(self, &file_options, file, Emited::Ast(ast))? {
1013            return finisher::archive_compilation_module_jit(self, file_time, file);
1014        }
1015
1016        if emit::before_frontend(self, &file_options, build_dir, file, Emited::Ast(ast))? {
1017            return finisher::archive_compilation_module_jit(self, file_time, file);
1018        }
1019
1020        if file_options.stop_compilation_at(CompilationPhase::Parser) {
1021            return finisher::archive_compilation_module_jit(self, file_time, file);
1022        }
1023
1024        {
1025            let mut semantic_analysis: SemanticAnalysis<'_> =
1026                SemanticAnalysis::new(ast, file, &file_options);
1027
1028            let semantic_analysis_failed: either::Either<bool, ()> =
1029                semantic_analysis.execute(parser_failed);
1030
1031            self.update_thrustc_frontend_time(frontend_time.elapsed());
1032
1033            match semantic_analysis_failed {
1034                either::Either::Left(semantic_analysis_failed) => {
1035                    if parser_failed || semantic_analysis_failed {
1036                        return interrupt::archive_compilation_module_jit(self, file, file_time);
1037                    }
1038                }
1039                either::Either::Right(()) => {
1040                    return finisher::archive_compilation_module_jit(self, file_time, file);
1041                }
1042            }
1043        }
1044
1045        {
1046            let mut intrinsic_checker: LLVMIntrinsicChecker<'_> =
1047                LLVMIntrinsicChecker::new(ast, file, self.options);
1048
1049            let mut call_conv_checker: LLVMCallConventionsChecker<'_> =
1050                LLVMCallConventionsChecker::new(ast, self.options, file);
1051
1052            let intrinsic_checker_fail: bool = intrinsic_checker.analyze();
1053
1054            if file_options.stop_compilation_at(CompilationPhase::LLVMIntrinsicChecker) {
1055                return finisher::archive_compilation_module_jit(self, file_time, file);
1056            }
1057
1058            let call_convention_fail: bool = call_conv_checker.analyze();
1059
1060            if file_options.stop_compilation_at(CompilationPhase::LLVMCallConventionChecker) {
1061                return finisher::archive_compilation_module_jit(self, file_time, file);
1062            }
1063
1064            self.update_thrustc_frontend_time(frontend_time.elapsed());
1065
1066            if intrinsic_checker_fail || call_convention_fail {
1067                return interrupt::archive_compilation_module_jit(self, file, file_time);
1068            }
1069        }
1070
1071        if print::after_frontend(self, &file_options, file, Emited::Ast(ast))? {
1072            return finisher::archive_compilation_module_jit(self, file_time, file);
1073        }
1074
1075        if emit::after_frontend(self, &file_options, build_dir, file, Emited::Ast(ast))? {
1076            return finisher::archive_compilation_module_jit(self, file_time, file);
1077        }
1078
1079        let backend_time: std::time::Instant = std::time::Instant::now();
1080
1081        let target: &LLVMTarget = llvm_backend.get_target();
1082        let llvm_triple: &TargetTriple = target.get_target_triple();
1083
1084        let llvm_target_triple_formatted: String =
1085            llvm_triple.as_str().to_string_lossy().to_string();
1086
1087        let llvm_target_triple: LLVMTargetTriple =
1088            LLVMTargetTriple::new(llvm_target_triple_formatted.clone());
1089
1090        let (llvm_cpu_name, llvm_cpu_features): (&str, &str) =
1091            llvm_backend.get_cross_target_cpu();
1092
1093        let compiler_optimization: ThrustOptimization = file_options.optimization();
1094        let llvm_opt: OptimizationLevel = compiler_optimization.to_llvm_opt();
1095
1096        let target: Target = Target::from_triple(llvm_triple).map_err(|_| {
1097            let _ = interrupt::archive_compilation_unit_with_message(
1098                self,
1099            thrustc_logging::LoggingType::Error,
1100                "Target-triple could not be built correctly. Maybe this target triple is invalid. Try another.",
1101                file,
1102                file_time,
1103            );
1104        })?;
1105
1106        let target_machine: TargetMachine = target
1107            .create_target_machine(
1108                llvm_triple,
1109                llvm_cpu_name,
1110                llvm_cpu_features,
1111                llvm_opt,
1112                file_options.reloc_model(),
1113                file_options.code_model(),
1114            )
1115            .ok_or_else(|| {
1116                let _ = interrupt::archive_compilation_unit_with_message(
1117                    self,
1118                    thrustc_logging::LoggingType::Error,
1119                    "Target machine could not be built correctly.",
1120                    file,
1121                    file_time,
1122                );
1123            })?;
1124
1125        let target_data: TargetData = target_machine.get_target_data();
1126        let target_triple: TargetTriple = target_machine.get_triple();
1127
1128        let target_info: TargetInfo =
1129            TargetInfo::new(LLVMTargetTriple::new(llvm_target_triple_formatted));
1130
1131        let mut target_abi: Option<LLVMABIRepresentation> = thrustc_llvm_abi::get_abi(
1132            self.options.abi_configuration().specific(),
1133            file,
1134            self.options,
1135            &llvm_target_triple,
1136            &target_info,
1137            &target_data,
1138        );
1139
1140        if self.options.abi_configuration().disable() {
1141            target_abi = None;
1142        }
1143
1144        let llvm_context: Context = Context::create();
1145        let llvm_builder: Builder = llvm_context.create_builder();
1146        let llvm_module: Module = llvm_context.create_module(file.get_name());
1147
1148        llvm_module.set_triple(llvm_triple);
1149        llvm_module.set_data_layout(&target_machine.get_target_data().get_data_layout());
1150
1151        jit::has_jit_available(&target)?;
1152
1153        let mut llvm_codegen_context: LLVMCodeGenContext = LLVMCodeGenContext::new(
1154            &llvm_module,
1155            &llvm_context,
1156            &llvm_builder,
1157            &target_data,
1158            &target_triple,
1159            &target_machine,
1160            target_abi.as_ref(),
1161            Diagnostician::new(file, self.options),
1162            self.options,
1163            &file_options,
1164            file,
1165        );
1166
1167        thrustc_llvm_codegen::LLVMCompiler::compile(&mut llvm_codegen_context, ast);
1168
1169        validate::llvm_codegen(&llvm_module, file)?;
1170
1171        self.update_thrustc_backend_time(backend_time.elapsed());
1172
1173        if print::llvm_before_optimization(
1174            self,
1175            &file_options,
1176            &llvm_module,
1177            &target_machine,
1178            file,
1179            file_time,
1180        )? {
1181            return finisher::archive_compilation_module_jit(self, file_time, file);
1182        }
1183
1184        if emit::llvm_before_optimization(
1185            self,
1186            &file_options,
1187            &llvm_module,
1188            &target_machine,
1189            build_dir,
1190            file,
1191            file_time,
1192        )? {
1193            return finisher::archive_compilation_module_jit(self, file_time, file);
1194        }
1195
1196        if file_options.stop_compilation_at(CompilationPhase::LLVMCodegen) {
1197            return finisher::archive_compilation_module_jit(self, file_time, file);
1198        }
1199
1200        let llvm_optimizer_config: LLVMOptimizationConfig = LLVMOptimizationConfig::new(
1201            compiler_optimization,
1202            file_options.sanitizer(),
1203            file_options.symbol_linkage_strategy(),
1204            file_options.denormal_fp(),
1205            file_options.denormal_fp_32(),
1206        );
1207
1208        let llvm_optimizer_passes: LLVMOptimizerPasses<'_> = LLVMOptimizerPasses::new(
1209            file_options.opt_passes(),
1210            file_options.modificator_opt_passes(),
1211        );
1212
1213        let llvm_optimizer_flags: LLVMOptimizerFlags = LLVMOptimizerFlags::new(
1214            file_options.omit_default_optimizations(),
1215            file_options.disable_all_sanitizers(),
1216        );
1217
1218        thrustc_llvm_codegen::optimizer::LLVMOptimizer::new(
1219            &llvm_module,
1220            &llvm_context,
1221            &target_machine,
1222            llvm_optimizer_config,
1223            llvm_optimizer_flags,
1224            llvm_optimizer_passes,
1225        )
1226        .optimize();
1227
1228        self.update_thrustc_backend_time(backend_time.elapsed());
1229
1230        if print::llvm_after_optimization(
1231            self,
1232            &file_options,
1233            &llvm_module,
1234            &target_machine,
1235            file,
1236            file_time,
1237        )? {
1238            return finisher::archive_compilation_module_jit(self, file_time, file);
1239        }
1240
1241        if emit::llvm_after_optimization(
1242            self,
1243            &file_options,
1244            &llvm_module,
1245            &target_machine,
1246            build_dir,
1247            file,
1248            file_time,
1249        )? {
1250            return finisher::archive_compilation_module_jit(self, file_time, file);
1251        }
1252
1253        self.update_thrustc_backend_time(backend_time.elapsed());
1254
1255        finisher::archive_compilation(self, file_time, file)?;
1256
1257        Ok(either::Either::Left(llvm_module.write_bitcode_to_memory()))
1258    }
1259}

1298impl ThrustCompiler<'_> {
1299    fn register_compiled_unit(&mut self, source: std::path::PathBuf, object: std::path::PathBuf) {
1300        let source: std::path::PathBuf = source.canonicalize().unwrap_or(source);
1301
1302        if let Some(previous) = self.source_to_object.insert(source, object.clone()) {
1303            self.linked_objects
1304                .retain(|candidate| candidate != &previous);
1305
1306            let _ = std::fs::remove_file(&previous);
1307        }
1308
1309        self.linked_objects.push(object);
1310    }
1311
1312    fn register_linked_sanitizer(&mut self, file_options: &FileOptions<'_, '_>) {
1313        if file_options.disable_all_sanitizers() {
1314            return;
1315        }
1316
1317        let sanitizer: Sanitizer = file_options.sanitizer();
1318
1319        if sanitizer.is_none() || self::contains_sanitizer_kind(&self.linked_sanitizers, sanitizer)
1320        {
1321            return;
1322        }
1323
1324        self.linked_sanitizers.push(sanitizer);
1325    }
