18use thrustc_ast::Ast;
19use thrustc_attributes::ThrustAttributes;
20use thrustc_builtins::BuiltinRegistry;
21use thrustc_code_location::Span;
22use thrustc_errors::CompilationIssue;
23use thrustc_options::{CompilationUnit, CompilerOptions};
24use thrustc_token::Token;
25use thrustc_token_type::TokenType;
26use thrustc_typesystem::Type;
27
28use crate::registry::SharedModuleRegistry;
29
30pub trait TypeParseContext {
31    fn peek(&mut self) -> &Token;
32    fn previous(&mut self) -> &Token;
33    fn advance(&mut self) -> Result<&Token, ()>;
34    fn only_advance(&mut self) -> Result<(), ()>;
35    fn consume(&mut self, kind: TokenType) -> Result<&Token, ()>;
36    fn consume_these(&mut self, these: &[TokenType]) -> Result<&Token, ()>;
37    fn check(&mut self, kind: TokenType) -> bool;
38    fn match_token(&mut self, kind: TokenType) -> Result<bool, ()>;
39
40    fn get_builtins(&self) -> &BuiltinRegistry;
41    fn get_registry(&self) -> SharedModuleRegistry;
42    fn get_options(&self) -> &CompilerOptions;
43    fn get_file(&self) -> &CompilationUnit;
44
45    fn resolve_named_type(&self, name: &str) -> Option<Type>;
46    fn resolve_type_parameter(&self, name: &str) -> Option<Span>;
47    fn resolve_named_generic(&self, name: &str) -> Option<(Vec<String>, Type)>;
48
49    fn enter_type(&mut self) -> Result<(), ()>;
50    fn leave_type(&mut self);
51
52    fn add_error(&mut self, error: CompilationIssue);
53
54    fn parse_constant_expr(&mut self) -> Result<Ast<'static>, ()>;
55    fn parse_attributes(&mut self, limits: &[TokenType]) -> Result<ThrustAttributes, ()>;
56}
