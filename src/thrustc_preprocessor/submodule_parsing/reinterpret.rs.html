20use thrustc_code_location::Span;
21use thrustc_typesystem::Type;
22
23pub fn floating_point(lexeme: &str, span: Span) -> Result<(Type, f64), ()> {
24    if lexeme.bytes().filter(|&b| b == b'.').count() > 1 {
25        Err(())
26    } else {
27        lexeme
28            .parse::<f32>()
29            .map(|f| (Type::F32 { span }, f as f64))
30            .or_else(|_| lexeme.parse::<f64>().map(|f| (Type::F64 { span }, f)))
31            .map_err(|_| ())
32    }
33}

35pub fn integer(lexeme: &str, span: Span) -> Result<(Type, u64), ()> {
36    const I8_MIN: isize = -128;
37    const I8_MAX: isize = 127;
38    const I16_MIN: isize = -32768;
39    const I16_MAX: isize = 32767;
40    const I32_MIN: isize = -2147483648;
41    const I32_MAX: isize = 2147483647;
42
43    const U8_MAX: usize = 255;
44    const U16_MAX: usize = 65535;
45    const U32_MAX: usize = 4294967295;
46
47    fn match_signed(number: isize, span: Span) -> Result<(Type, u64), ()> {
48        match number {
49            n if (I8_MIN..=I8_MAX).contains(&n) => Ok((Type::S8 { span }, n as u64)),
50            n if (I16_MIN..=I16_MAX).contains(&n) => Ok((Type::S16 { span }, n as u64)),
51            n if (I32_MIN..=I32_MAX).contains(&n) => Ok((Type::S32 { span }, n as u64)),
52            n if (isize::MIN..=isize::MAX).contains(&n) => Ok((Type::S64 { span }, n as u64)),
53
54            _ => Err(()),
55        }
56    }
57
58    fn match_unsigned(number: usize, span: Span) -> Result<(Type, u64), ()> {
59        match number {
60            n if (0..=U8_MAX).contains(&n) => Ok((Type::U8 { span }, n as u64)),
61            n if (0..=U16_MAX).contains(&n) => Ok((Type::U16 { span }, n as u64)),
62            n if (0..=U32_MAX).contains(&n) => Ok((Type::U32 { span }, n as u64)),
63            n if (0..=usize::MAX).contains(&n) => Ok((Type::U64 { span }, n as u64)),
64
65            _ => Err(()),
66        }
67    }
68
69    let hexadecimal: bool = lexeme.strip_prefix("0x").is_some();
70    let octal: bool = lexeme.strip_prefix("0o").is_some();
71    let binary: bool = lexeme.strip_prefix("0b").is_some();
72
73    let (radix, prefix) = if hexadecimal {
74        (16, "0x")
75    } else if octal {
76        (8, "0o")
77    } else if binary {
78        (2, "0b")
79    } else {
80        (10, "")
81    };
82
83    let cleaned: String = if radix != 10 {
84        lexeme
85            .strip_prefix(prefix)
86            .unwrap_or(lexeme)
87            .replace('_', "")
88    } else {
89        lexeme.replace('_', "")
90    };
91
92    if radix != 10 {
93        if let Ok(n) = usize::from_str_radix(&cleaned, radix) {
94            return match_unsigned(n, span);
95        }
96
97        if let Ok(n) = isize::from_str_radix(&cleaned, radix) {
98            return match_signed(n, span);
99        }
100
101        Err(())
102    } else {
103        if let Ok(n) = lexeme.parse::<usize>() {
104            return match_unsigned(n, span);
105        }
106
107        if let Ok(n) = lexeme.parse::<isize>() {
108            return match_signed(n, span);
109        }
110
111        Err(())
112    }
113}
