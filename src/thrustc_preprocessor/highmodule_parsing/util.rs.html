20use std::path::PathBuf;
21
22pub fn verbose_include_dirs(command: &str) -> Vec<PathBuf> {
23    let mut dirs: Vec<PathBuf> = Vec::new();
24
25    let output: std::process::Output = match std::process::Command::new(command)
26        .args(["-E", "-xc", "-v", "-"])
27        .stdin(std::process::Stdio::null())
28        .output()
29    {
30        Ok(output) => output,
31        Err(_) => return dirs,
32    };
33
34    let stderr: std::borrow::Cow<'_, str> = String::from_utf8_lossy(&output.stderr);
35
36    let mut tokens: Vec<String> = Vec::new();
37
38    for token in stderr.split_whitespace() {
39        tokens.push(token.to_string());
40    }
41
42    let mut idx: usize = 0;
43
44    while idx < tokens.len() {
45        if tokens[idx] == "-internal-isystem" || tokens[idx] == "-internal-externc-isystem" {
46            if let Some(dir) = tokens.get(idx.saturating_add(1)) {
47                dirs.push(PathBuf::from(dir));
48            }
49
50            idx = idx.saturating_add(2);
51        } else {
52            idx = idx.saturating_add(1);
53        }
54    }
55
56    let mut in_list: bool = false;
57
58    for line in stderr.lines() {
59        let trimmed: &str = line.trim();
60
61        if trimmed == "#include <...> search starts here:" {
62            in_list = true;
63        } else if trimmed == "End of search list." {
64            in_list = false;
65        } else if in_list && !trimmed.is_empty() {
66            dirs.push(PathBuf::from(trimmed));
67        }
68    }
69
70    dirs
71}

73pub fn detect_host_system_include_dirs() -> Vec<PathBuf> {
74    let mut collected: Vec<PathBuf> = Vec::new();
75
76    let mut push_unique = |candidate: PathBuf| {
77        let canonical: PathBuf = candidate
78            .canonicalize()
79            .unwrap_or_else(|_| candidate.clone());
80
81        let already: bool = collected.iter().any(|known: &PathBuf| {
82            known.canonicalize().unwrap_or_else(|_| known.clone()) == canonical
83        });
84
85        if !already && canonical.is_dir() {
86            collected.push(canonical);
87        }
88    };
89
90    let from_verbose: Vec<PathBuf> = self::clang_commands()
91        .into_iter()
92        .map(|command| self::verbose_include_dirs(&command))
93        .find(|dirs| !dirs.is_empty())
94        .unwrap_or_default();
95
96    from_verbose.into_iter().for_each(&mut push_unique);
97
98    let env_iter = ["C_INCLUDE_PATH", "CPATH", "INCLUDE"]
99        .iter()
100        .filter_map(std::env::var_os)
101        .flat_map(|paths| {
102            std::env::split_paths(&paths)
103                .filter(|entry| !entry.as_os_str().is_empty())
104                .collect::<Vec<PathBuf>>()
105        });
106
107    env_iter.for_each(&mut push_unique);
108
109    let fallback_dirs: Vec<PathBuf> = if cfg!(windows) {
110        Vec::new()
111    } else {
112        ["/usr/local/include", "/usr/include"]
113            .iter()
114            .map(PathBuf::from)
115            .collect()
116    };
117
118    fallback_dirs.into_iter().for_each(&mut push_unique);
119
120    collected
121}

123pub fn detect_clang_resource_include_dir() -> Option<PathBuf> {
124    let commands: Vec<String> = self::clang_commands();
125
126    let outputs = commands.iter().filter_map(|command| {
127        std::process::Command::new(command)
128            .arg("-print-resource-dir")
129            .output()
130            .ok()
131    });
132
133    let successful = outputs.filter(|output| output.status.success());
134
135    let mut include_dirs = successful.filter_map(|output| {
136        let resource_dir: std::borrow::Cow<'_, str> = String::from_utf8_lossy(&output.stdout);
137
138        let trimmed: &str = resource_dir.trim();
139
140        if trimmed.is_empty() {
141            None
142        } else {
143            Some(PathBuf::from(trimmed).join("include"))
144        }
145    });
146
147    include_dirs.find(|include_dir| include_dir.is_dir())
148}

150pub fn clang_commands() -> Vec<String> {
151    let env_opt: Option<String> = std::env::var("CLANG")
152        .ok()
153        .filter(|value| !value.trim().is_empty());
154
155    let mut commands: Vec<String> = env_opt.into_iter().collect();
156
157    let fallback_iter = [
158        "clang", "clang-16", "clang-17", "clang-18", "clang-19", "clang-20", "clang-21",
159        "clang-22", "clang-23", "cc",
160    ]
161    .iter()
162    .map(|name| name.to_string());
163
164    commands.extend(fallback_iter);
165
166    commands
167}
