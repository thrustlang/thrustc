125pub fn parse_import_c<'preprocessor>(
126    parser: &mut PreprocessorContext<'preprocessor>,
127) -> Result<Option<Module>, ()> {
128    let request: ImportC = self::parse_import_request(parser)?;
129
130    let base_name: String = Path::new(if request.outside_header() {
131        request.import_path()
132    } else {
133        request.header_path()
134    })
135    .file_stem()
136    .map_or_else(String::new, |stem| stem.to_string_lossy().to_string());
137
138    let mut module: Module = Module::new(base_name, request.header_path().clone());
139
140    if let Some(alias) = request.alias().clone() {
141        module.set_alias(alias);
142    }
143
144    let options: thrustc_import_c::options::CImportOptions = self::build_import_options(
145        request.import_opts(),
146        request.outside_header(),
147        request.resolved_from_import_search(),
148    );
149
150    let context: thrustc_import_c::context::CImportContext =
151        self::synthesize_imported_header(parser, &request, options)?;
152
153    self::emit_import_diagnostics(parser, &context, request.span());
154    self::emit_imported_symbols(&mut module, &context, request.span());
155
156    parser.get_registry().borrow_mut().register(&module);
157
158    Ok(Some(module))
159}

161fn emit_import_diagnostics(
162    parser: &mut PreprocessorContext,
163    context: &thrustc_import_c::context::CImportContext,
164    span: Span,
165) {
166    for diagnostic in context.diagnostics() {
167        let code: CompilationIssueCode = match diagnostic.kind() {
168            thrustc_import_c::diagnostics::CImportDiagnosticKind::ClangDiagnostic => {
169                CompilationIssueCode::W0100
170            }
171            thrustc_import_c::diagnostics::CImportDiagnosticKind::InventedFieldName => {
172                CompilationIssueCode::W0101
173            }
174            thrustc_import_c::diagnostics::CImportDiagnosticKind::SkippedUnion => {
175                CompilationIssueCode::W0102
176            }
177            thrustc_import_c::diagnostics::CImportDiagnosticKind::SkippedBitfieldStruct => {
178                CompilationIssueCode::W0103
179            }
180            thrustc_import_c::diagnostics::CImportDiagnosticKind::UnsupportedCallingConvention => {
181                CompilationIssueCode::E0100
182            }
183            thrustc_import_c::diagnostics::CImportDiagnosticKind::SkippedDeclaration
184            | thrustc_import_c::diagnostics::CImportDiagnosticKind::SkippedMacro
185            | thrustc_import_c::diagnostics::CImportDiagnosticKind::UnsupportedMacro => {
186                CompilationIssueCode::W0104
187            }
188        };
189
190        if diagnostic.kind()
191            == thrustc_import_c::diagnostics::CImportDiagnosticKind::UnsupportedCallingConvention
192        {
193            parser.add_error(CompilationIssue::Error(
194                code,
195                diagnostic.message().to_string(),
196                diagnostic
197                    .help()
198                    .unwrap_or(
199                        "Use a function with a calling convention that Thrust can import, or remove the importC dependency on that declaration.",
200                    )
201                    .to_string(),
202                diagnostic.note().map(str::to_string),
203                diagnostic.span().unwrap_or(span),
204            ));
205        } else {
206            let mut message: String = diagnostic.message().to_string();
207
208            if let Some(help) = diagnostic.help() {
209                message.push_str("\nhelp: ");
210                message.push_str(help);
211            }
212
213            if let Some(note) = diagnostic.note() {
214                message.push_str("\nnote: ");
215                message.push_str(note);
216            }
217
218            parser.add_warning(CompilationIssue::Warning(
219                code,
220                message,
221                diagnostic.span().unwrap_or(span),
222            ));
223        }
224    }
225}

227fn emit_imported_symbols(
228    module: &mut Module,
229    context: &thrustc_import_c::context::CImportContext,
230    span: Span,
231) {
232    for macro_definition in context.macros() {
233        module.add_macro(macro_definition.clone());
234    }
235
236    for function in context.functions() {
237        let name: String = function.name().to_string();
238
239        let mut parameters: Vec<(String, Type, Span)> =
240            Vec::with_capacity(function.parameter_types().len());
241
242        for (param_name, param_type) in function
243            .parameter_names()
244            .iter()
245            .cloned()
246            .zip(function.parameter_types().iter().cloned())
247        {
248            parameters.push((param_name, param_type, span));
249        }
250
251        let mut attributes: thrustc_attributes::ThrustAttributes = Vec::with_capacity(4);
252
253        attributes.push(ThrustAttribute::Convention(
254            function.convention().to_string(),
255            span,
256        ));
257
258        if function.variadic() {
259            attributes.push(ThrustAttribute::Ignore(span));
260            attributes.push(ThrustAttribute::NoArgCount(span));
261        }
262
263        let symbol: Symbol = Symbol {
264            name,
265            signature: Signature::Function {
266                kind: function.return_type().clone(),
267                invalid_kind: Type::Void { span },
268                demangling_name: function.external_name().to_string(),
269                type_params: None,
270                parameters,
271                attributes,
272                span,
273            },
274            variant: Variant::Function,
275            public: true,
276        };
277
278        module.add_symbol(symbol);
279    }
280
281    for record in context.structs() {
282        let name: String = record.name().to_string();
283
284        let mut fields: Vec<(String, Type, Span)> = Vec::with_capacity(record.fields().len());
285        let mut field_types: Vec<Type> = Vec::with_capacity(record.fields().len());
286
287        for (field_name, field_type) in record.fields().iter() {
288            fields.push((field_name.clone(), field_type.clone(), span));
289            field_types.push(field_type.clone());
290        }
291
292        let metadata: StructTypeMetadata = *record.metadata();
293
294        let kind: Type = Type::Struct {
295            name: name.clone(),
296            fields: field_types,
297            metadata,
298            span,
299        };
300
301        let symbol: Symbol = Symbol {
302            name,
303            signature: Signature::Struct {
304                kind,
305                invalid_kind: Type::Void { span },
306                type_params: None,
307                fields,
308                span,
309            },
310            variant: Variant::Struct,
311            public: true,
312        };
313
314        module.add_symbol(symbol);
315    }
316
317    for imported_enum in context.enums() {
318        let name: String = imported_enum.name().to_string();
319        let underlying_type: Type = imported_enum.underlying_type().clone();
320
321        let mut fields: Vec<(
322            String,
323            Type,
324            Option<thrustc_compile_time::BuiltinValue>,
325            Span,
326        )> = Vec::with_capacity(imported_enum.fields().len());
327
328        for (field_name, value) in imported_enum.fields().iter() {
329            fields.push((
330                field_name.clone(),
331                underlying_type.clone(),
332                Some(thrustc_compile_time::BuiltinValue::Integer(*value)),
333                span,
334            ));
335        }
336
337        let symbol: Symbol = Symbol {
338            name,
339            signature: Signature::Enum {
340                invalid_kind: Type::Void { span },
341                fields,
342                attributes: Vec::new(),
343                span,
344            },
345            variant: Variant::Enum,
346            public: true,
347        };
348
349        module.add_symbol(symbol);
350
351        for (field_name, value) in imported_enum.fields().iter() {
352            if module
353                .get_symbols()
354                .iter()
355                .any(|symbol| symbol.name == *field_name)
356            {
357                continue;
358            }
359
360            module.add_symbol(Symbol {
361                name: field_name.clone(),
362                signature: Signature::Constant {
363                    kind: underlying_type.clone(),
364                    invalid_kind: Type::Void { span },
365                    value: Some(thrustc_compile_time::BuiltinValue::Integer(*value)),
366                    attributes: Vec::new(),
367                    modificators: Modificators::new(),
368                    span,
369                },
370                variant: Variant::Constant,
371                public: true,
372            });
373        }
374    }
375
376    for imported_type in context.typedefs() {
377        let name: String = imported_type.name().to_string();
378        let ty: Type = imported_type.ty().clone();
379
380        let symbol: Symbol = Symbol {
381            name,
382            signature: Signature::CustomType {
383                kind: ty,
384                invalid_kind: Type::Void { span },
385                type_params: None,
386                attributes: Vec::new(),
387                span,
388            },
389            variant: Variant::CustomType,
390            public: true,
391        };
392
393        module.add_symbol(symbol);
394    }
395
396    for imported_static in context.statics() {
397        let name: String = imported_static.name().to_string();
398        let kind: Type = imported_static.kind().clone();
399
400        let attributes: thrustc_attributes::ThrustAttributes = vec![
401            ThrustAttribute::Public(span),
402            ThrustAttribute::Extern(imported_static.external_name().to_string(), span),
403        ];
404
405        let symbol: Symbol = Symbol {
406            name,
407            signature: Signature::Static {
408                kind,
409                invalid_kind: Type::Void { span },
410                is_mutable: imported_static.is_mutable(),
411                attributes,
412                modificators: Modificators::new(),
413                span,
414            },
415            variant: Variant::Static,
416            public: true,
417        };
418
419        module.add_symbol(symbol);
420    }
421
422    for constant in context.constants() {
423        let name: String = constant.name().to_string();
424        let kind: Type = constant.kind().clone();
425        let value: thrustc_compile_time::BuiltinValue = constant.value().clone();
426
427        let symbol: Symbol = Symbol {
428            name,
429            signature: Signature::Constant {
430                kind,
431                invalid_kind: Type::Void { span },
432                value: Some(value),
433                attributes: Vec::new(),
434                modificators: Modificators::new(),
435                span,
436            },
437            variant: Variant::Constant,
438            public: true,
439        };
440
441        module.add_symbol(symbol);
442    }
443}

445fn parse_import_request(parser: &mut PreprocessorContext) -> Result<ImportC, ()> {
446    parser.consume(TokenType::ImportC)?;
447
448    let current_path: PathBuf = parser.get_compilation_unit().get_path().to_path_buf();
449
450    let current_dir: PathBuf = current_path
451        .parent()
452        .map_or_else(|| PathBuf::from("."), |p| p.to_path_buf());
453
454    let header_path_tk: &Token =
455        parser.consume_these(&[TokenType::CString, TokenType::CNString])?;
456
457    let import_str: String = header_path_tk
458        .get_lexeme()
459        .trim()
460        .trim_matches('"')
461        .to_string();
462
463    let span: Span = header_path_tk.get_code_location();
464
465    let import_path: PathBuf = PathBuf::from(&import_str);
466    let mut header_path: PathBuf = import_path.clone();
467    let mut outside_header: bool = false;
468    let mut resolved_from_import_search: bool = false;
469    let mut parse_outside_header: bool = false;
470
471    if parser.check(TokenType::Only) {
472        let only_span: Span = parser.peek().get_code_location();
473
474        parser.add_error(CompilationIssue::Error(
475            CompilationIssueCode::E0035,
476            "The 'only' list is not supported by importC.".into(),
477            "Remove the 'only { ... }' clause.".into(),
478            None,
479            only_span,
480        ));
481
482        return Err(());
483    }
484
485    let mut alias: Option<Vec<String>> = None;
486
487    if parser.check(TokenType::As) {
488        parser.consume(TokenType::As)?;
489
490        let alias_tk: &Token = parser.consume(TokenType::Identifier)?;
491        let mut alias_parts: Vec<String> = vec![alias_tk.get_lexeme().to_string()];
492
493        while parser.check(TokenType::ColonColon) {
494            parser.only_advance()?;
495
496            let part_tk: &Token = parser.consume(TokenType::Identifier)?;
497
498            alias_parts.push(part_tk.get_lexeme().to_string());
499        }
500
501        alias = Some(alias_parts);
502    }
503
504    parser.consume(TokenType::SemiColon)?;
505
506    let file_options: &thrustc_directive::FileOptions<'_, '_> = parser.get_file_options();
507    let import_opts: thrustc_options::ImportCOptions =
508        thrustc_directive::combine_import_c_options(file_options);
509
510    let header_is_relative: bool = header_path.is_relative();
511
512    let local_candidate: PathBuf = current_dir.join(&import_str);
513
514    let local_exists: bool = header_is_relative && local_candidate.exists();
515
516    if local_exists {
517        header_path = local_candidate;
518    }
519
520    let should_search_import_paths: bool = header_is_relative && !local_exists;
521
522    let found: Option<PathBuf> = if should_search_import_paths {
523        import_opts
524            .include_paths()
525            .iter()
526            .chain(import_opts.system_include_paths().iter())
527            .map(|include_dir| include_dir.join(&import_str))
528            .find(|candidate| candidate.exists())
529    } else {
530        None
531    };
532
533    let found_is_missing: bool = found.is_none();
534
535    if let Some(candidate) = found {
536        header_path = candidate;
537        resolved_from_import_search = true;
538        parse_outside_header = true;
539    }
540
541    if should_search_import_paths && found_is_missing {
542        outside_header = true;
543    }
544
545    if !outside_header && let Ok(canonicalized) = header_path.canonicalize() {
546        header_path = canonicalized;
547    }
548
549    let mut current_file_path: PathBuf = current_path.clone();
550
551    if let Ok(canonicalized_current) = current_path.canonicalize() {
552        current_file_path = canonicalized_current;
553    }
554
555    if !outside_header && header_path == current_file_path {
556        parser.add_error(CompilationIssue::Error(
557            CompilationIssueCode::E0035,
558            "The header cannot be imported itself.".into(),
559            "You should remove it.".into(),
560            None,
561            span,
562        ));
563
564        return Err(());
565    }
566
567    if !outside_header && !header_path.exists() {
568        parser.add_error(CompilationIssue::Error(
569            CompilationIssueCode::E0035,
570            "The path does not exist.".into(),
571            "Use an existing local header path or pass '--import-c-include <dir>' / '--import-c-system-include <dir>' so Clang can resolve the outside header.".into(),
572            None,
573            span,
574        ));
575
576        return Err(());
577    }
578
579    if !outside_header && !header_path.is_file() {
580        parser.add_error(CompilationIssue::Error(
581            CompilationIssueCode::E0035,
582            "The path does not point to a file.".into(),
583            "You should make sure it is a valid path to file.".into(),
584            None,
585            span,
586        ));
587
588        return Err(());
589    }
590
591    let import_c: ImportC = ImportC::new(
592        import_str,
593        import_path,
594        header_path,
595        outside_header,
596        resolved_from_import_search,
597        parse_outside_header,
598        alias,
599        import_opts,
600        span,
601    );
602
603    Ok(import_c)
604}

606fn build_import_options(
607    import_opts: &thrustc_options::ImportCOptions,
608    outside_header: bool,
609    resolved_from_import_search: bool,
610) -> thrustc_import_c::options::CImportOptions {
611    let mut options: thrustc_import_c::options::CImportOptions =
612        thrustc_import_c::options::CImportOptions::new();
613
614    let effective_scope: thrustc_options::ImportCScope = if import_opts.import_scope_overridden() {
615        import_opts.import_scope()
616    } else if outside_header || resolved_from_import_search {
617        thrustc_options::ImportCScope::TransitiveAll
618    } else {
619        thrustc_options::ImportCScope::MainOnly
620    };
621
622    let import_scope: thrustc_import_c::options::CImportScope = match effective_scope {
623        thrustc_options::ImportCScope::MainOnly => {
624            thrustc_import_c::options::CImportScope::MainOnly
625        }
626        thrustc_options::ImportCScope::TransitiveNoSystem => {
627            thrustc_import_c::options::CImportScope::TransitiveNoSystem
628        }
629        thrustc_options::ImportCScope::TransitiveAll => {
630            thrustc_import_c::options::CImportScope::TransitiveAll
631        }
632    };
633
634    options.set_import_scope(import_scope);
635
636    // Build clang arguments from `--import-c-*` flags.
637    {
638        let clang_args: &mut Vec<String> = options.clang_args_mut();
639
640        if let Some(resource_dir) =
641            crate::highmodule_parsing::util::detect_clang_resource_include_dir()
642        {
643            clang_args.push(format!("-isystem{}", resource_dir.display()));
644        }
645
646        for inc in import_opts.include_paths() {
647            clang_args.push(format!("-I{}", inc.display()));
648        }
649
650        for inc in import_opts.system_include_paths() {
651            clang_args.push(format!("-isystem{}", inc.display()));
652        }
653
654        let auto_system: Vec<PathBuf> = if import_opts.system_include_paths().is_empty() {
655            crate::highmodule_parsing::util::detect_host_system_include_dirs()
656        } else {
657            Vec::new()
658        };
659
660        let auto_iter = auto_system
661            .iter()
662            .map(|dir| format!("-isystem{}", dir.display()))
663            .filter(|flag| !clang_args.contains(flag));
664
665        let auto_flags: Vec<String> = auto_iter.collect();
666
667        clang_args.extend(auto_flags);
668
669        for def in import_opts.defines() {
670            clang_args.push(format!("-D{def}"));
671        }
672
673        for und in import_opts.undefs() {
674            clang_args.push(format!("-U{und}"));
675        }
676
677        if let Some(target) = import_opts.target() {
678            clang_args.push(format!("--target={target}"));
679        }
680
681        if let Some(sysroot) = import_opts.sysroot() {
682            clang_args.push(format!("--sysroot={}", sysroot.display()));
683        }
684
685        if let Some(std_) = import_opts.std() {
686            clang_args.push(format!("-std={std_}"));
687        }
688
689        clang_args.extend(import_opts.args().iter().cloned());
690    }
691
692    options
693}

695fn synthesize_imported_header(
696    parser: &mut PreprocessorContext,
697    import_c: &ImportC,
698    options: thrustc_import_c::options::CImportOptions,
699) -> Result<thrustc_import_c::context::CImportContext, ()> {
700    let importer_header_path: PathBuf = if import_c.parse_outside_header() {
701        import_c.import_path().clone()
702    } else {
703        import_c.header_path().clone()
704    };
705
706    let mut context: thrustc_import_c::context::CImportContext =
707        thrustc_import_c::context::CImportContext::new(
708            importer_header_path,
709            import_c.span(),
710            options,
711        );
712
713    if let Err(message) = context.import_header() {
714        let outside_header: bool = import_c.outside_header();
715
716        let import_str: &String = import_c.import_str();
717
718        let header_display: String = if outside_header {
719            import_str.clone()
720        } else {
721            import_c.header_path().display().to_string()
722        };
723
724        let help: String = if outside_header {
725            "Pass '--import-c-system-include <dir>' for system headers or '--import-c-include <dir>' for local/vendor headers so importC can resolve the header.".into()
726        } else {
727            "Pass '--import-c-system-include <dir>' or '--import-c-include <dir>' so Clang can resolve the header and its dependencies.".into()
728        };
729
730        let has_diagnostics: bool = !context.diagnostics().is_empty();
731
732        let looks_like_outside_header: bool =
733            import_str.ends_with(".h") && !import_str.contains(std::path::MAIN_SEPARATOR);
734
735        let diagnostic_count: usize = context.diagnostics().len();
736
737        let joined: String = context
738            .diagnostics()
739            .iter()
740            .map(|diagnostic| diagnostic.message().to_string())
741            .collect::<Vec<String>>()
742            .join("\n");
743
744        let message_text: String = if joined.is_empty() {
745            format!("Failed to import C header '{header_display}'.")
746        } else {
747            format!("Failed to import C header '{header_display}':\n{joined}")
748        };
749
750        let hint: Option<String> = match (has_diagnostics, outside_header, looks_like_outside_header) {
751            (true, _, _) if outside_header => Some(
752                "Try '--import-c-system-include <dir>' for system headers or '--import-c-include <dir>' for local/vendor headers.".into(),
753            ),
754            (true, _, _) => None,
755            (false, false, _) => Some(
756                "System headers may be filtered by the configured import scope. Try '--import-c-scope=transitive-all' if you need them."
757                    .into(),
758            ),
759            (false, true, true) => Some(
760                "This looks like an outside header. Try '--import-c-system-include <dir>' for system headers or '--import-c-include <dir>' for local/vendor headers."
761                    .into(),
762            ),
763            (false, true, false) => None,
764        };
765
766        let extra: String = format!("{diagnostic_count} clang diagnostic(s) reported.");
767
768        let note: Option<String> = match hint {
769            Some(hint) => Some(format!("{extra} {hint}")),
770            None if diagnostic_count > 0 => Some(extra),
771            None => Some(message),
772        };
773
774        parser.add_error(CompilationIssue::Error(
775            CompilationIssueCode::E0100,
776            message_text,
777            help,
778            note,
779            import_c.span(),
780        ));
781
782        return Err(());
783    }
784
785    Ok(context)
786}
