18use thrustc_code_location::Span;
19use thrustc_errors::CompilationIssue;
20use thrustc_errors::CompilationIssueCode;
21use thrustc_typesystem::Type;
22
23use crate::context::BuiltinContext;
24use crate::traits::BuiltinFunctionSignature;
25use crate::traits::BuiltinParameter;
26use crate::traits::CompileTimeBuiltinFunction;
27use crate::value::BuiltinArgument;
28use crate::value::BuiltinValue;
29
30#[derive(Debug)]
31pub struct SizeOf;
32
33impl CompileTimeBuiltinFunction for SizeOf {
34    fn name(&self) -> &'static str {
35        "sizeOf"
36    }
37
38    fn signature(&self) -> BuiltinFunctionSignature {
39        BuiltinFunctionSignature {
40            return_type: Type::USize {
41                span: Span::nothing(),
42            },
43            parameters: vec![BuiltinParameter::Type],
44        }
45    }
46
47    fn evaluate(
48        &self,
49        args: &[BuiltinArgument],
50        context: &mut BuiltinContext<'_>,
51    ) -> Result<BuiltinValue, CompilationIssue> {
52        let ty: &Type = match &args[0] {
53            BuiltinArgument::Type { ty, .. } => ty,
54            BuiltinArgument::Value { span, .. } => {
55                return Err(CompilationIssue::Error(
56                    CompilationIssueCode::E0019,
57                    "The 'sizeOf' compiler builtin expects a type argument.".into(),
58                    "You should pass a type to 'sizeOf', like sizeOf(u32).".into(),
59                    None,
60                    *span,
61                ));
62            }
63        };
64
65        let size_of: u32 = match context.target_info.get_type_layout(ty) {
66            either::Either::Left(layout) => layout.into_layout().sizeof,
67            either::Either::Right(layout) => layout.into_layout().sizeof,
68        };
69
70        Ok(BuiltinValue::Integer(size_of as u64))
71    }
72}
