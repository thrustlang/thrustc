20use thrustc_code_location::Span;
21use thrustc_typesystem::Type;
22
23#[derive(Debug)]
24pub struct MacroType;
25
26impl MacroType {
27    pub fn from_thrust_text(text: &str, span: Span) -> Type {
28        match text {
29            "s8" => Type::S8 { span },
30            "s16" => Type::S16 { span },
31            "s32" => Type::S32 { span },
32            "s64" => Type::S64 { span },
33            "ssize" => Type::SSize { span },
34            "u8" => Type::U8 { span },
35            "u16" => Type::U16 { span },
36            "u32" => Type::U32 { span },
37            "u64" => Type::U64 { span },
38            "u128" => Type::U128 { span },
39            "usize" => Type::USize { span },
40            "f32" => Type::F32 { span },
41            "f64" => Type::F64 { span },
42            "f128" => Type::F128 { span },
43            "bool" => Type::Bool { span },
44            "char" => Type::Char { span },
45            "void" => Type::Void { span },
46            _ => Type::Unresolved {
47                hint: text.to_string(),
48                span,
49            },
50        }
51    }
52
53    pub fn to_thrust_text(ty: &Type) -> String {
54        crate::type_format::format_type_thrust(ty)
55    }
56}
