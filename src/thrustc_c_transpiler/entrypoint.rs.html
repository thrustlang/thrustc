28pub fn handle_emit_c_bindings_thrust(
29    header: Option<PathBuf>,
30    out_dir: Option<PathBuf>,
31    output: Option<PathBuf>,
32    import_opts: &thrustc_options::ImportCOptions,
33) {
34    let Some(header) = header else {
35        return;
36    };
37
38    let mut emit_opts: crate::options::EmitCBindingsOptions =
39        crate::options::EmitCBindingsOptions::new();
40
41    if let Some(out_dir) = out_dir {
42        emit_opts.set_out_dir(out_dir);
43    }
44
45    if let Some(output) = output {
46        emit_opts.set_output(output);
47    }
48
49    match crate::manager::emit_c_bindings_thrust(header, import_opts, &emit_opts) {
50        Ok((out_path, contents, warnings)) => {
51            if let Some(parent) = out_path.parent() {
52                let _ = std::fs::create_dir_all(parent);
53            }
54
55            let mut open_options: std::fs::OpenOptions = std::fs::File::options();
56
57            open_options.create(true);
58            open_options.truncate(true);
59            open_options.write(true);
60
61            if let Ok(mut file) = open_options.open(&out_path) {
62                let _ = file.write_all(contents.as_bytes());
63            } else {
64                thrustc_logging::print_critical_error(
65                    LoggingType::Error,
66                    &format!("Unable to write output '{}'.", out_path.display()),
67                );
68            }
69
70            for warning in warnings {
71                if let CompilationIssue::Warning(code, message, ..) = warning {
72                    thrustc_logging::print_warning(
73                        LoggingType::Warning,
74                        &format!("{}: {}", code.to_title(), message),
75                    );
76                }
77            }
78
79            thrustc_logging::write(
80                OutputIn::Stdout,
81                &format!("Emitted C bindings to '{}'.\n", out_path.display()),
82            );
83
84            std::process::exit(thrustc_constants::SUCCESFUL_CODE);
85        }
86        Err(message) => {
87            thrustc_logging::print_critical_error(LoggingType::Error, &message);
88        }
89    }
90}

92pub fn handle_translate_c_to_thrust(inputs: &[PathBuf], translate_opts: &TranslateCOptions) {
93    if inputs.is_empty() {
94        return;
95    }
96
97    match self::translate_c_to_thrust(inputs, translate_opts) {
98        Ok(results) => {
99            for (out_path, contents, issues) in results {
100                if let Some(parent) = out_path.parent() {
101                    let _ = std::fs::create_dir_all(parent);
102                }
103
104                let mut open_options: std::fs::OpenOptions = std::fs::File::options();
105
106                open_options.create(true);
107                open_options.truncate(true);
108                open_options.write(true);
109
110                if let Ok(mut file) = open_options.open(&out_path) {
111                    let _ = file.write_all(contents.as_bytes());
112                } else {
113                    thrustc_logging::print_critical_error(
114                        LoggingType::Error,
115                        &format!("Unable to write output '{}'.", out_path.display()),
116                    );
117                }
118
119                for issue in issues {
120                    match issue {
121                        CompilationIssue::Warning(code, message, ..) => {
122                            thrustc_logging::print_warning(
123                                LoggingType::Warning,
124                                &format!("{}: {}\n", code.to_title(), message),
125                            );
126                        }
127                        CompilationIssue::Error(code, message, help, note, ..) => {
128                            let mut full: String =
129                                format!("{}: {}\nhelp: {}", code.to_title(), message, help);
130
131                            if let Some(note) = note {
132                                full.push_str("\nnote: ");
133                                full.push_str(&note);
134                            }
135
136                            thrustc_logging::print_error(LoggingType::Error, &full);
137                        }
138                        _ => {}
139                    }
140                }
141            }
142
143            thrustc_logging::write(OutputIn::Stdout, "C successfully translated to Thrust.\n");
144
145            std::process::exit(thrustc_constants::SUCCESFUL_CODE);
146        }
147
148        Err(issues) => {
149            for issue in issues {
150                match issue {
151                    CompilationIssue::Warning(code, message, ..) => {
152                        thrustc_logging::print_warning(
153                            LoggingType::Warning,
154                            &format!("{}: {}\n", code.to_title(), message),
155                        );
156                    }
157                    CompilationIssue::Error(code, message, help, note, ..) => {
158                        let mut full: String =
159                            format!("{}: {}\nhelp: {}", code.to_title(), message, help);
160
161                        if let Some(note) = note {
162                            full.push_str("\nnote: ");
163                            full.push_str(&note);
164                        }
165
166                        thrustc_logging::print_error(LoggingType::Error, &full);
167                    }
168                    _ => {}
169                }
170            }
171
172            std::process::exit(thrustc_constants::FAILURE_CODE);
173        }
174    }
175}

177pub fn translate_c_to_thrust(
178    inputs: &[PathBuf],
179    translate_opts: &TranslateCOptions,
180) -> Result<Vec<TranslateCOutput>, Vec<CompilationIssue>> {
181    let mut outputs: Vec<TranslateCOutput> = Vec::new();
182
183    let mut errors: Vec<CompilationIssue> = Vec::new();
184
185    if inputs.is_empty() {
186        errors.push(CompilationIssue::Error(
187            CompilationIssueCode::E0110,
188            "No input C files were provided.".into(),
189            "Pass at least one '.c' file.".into(),
190            None,
191            thrustc_code_location::Span::nothing(),
192        ));
193
194        return Err(errors);
195    }
196
197    if inputs.len() > 1 && translate_opts.output().is_some() {
198        errors.push(CompilationIssue::Error(
199            CompilationIssueCode::E0110,
200            "'--translate-c-output' only supports a single input file.".into(),
201            "Use '--translate-c-out-dir' or omit '--translate-c-output' when translating multiple inputs.".into(),
202            None,
203            thrustc_code_location::Span::nothing(),
204        ));
205
206        return Err(errors);
207    }
208
209    for input in inputs {
210        match crate::manager::translate_single_c_to_thrust(input, translate_opts) {
211            Ok(result) => outputs.push(result),
212            Err(mut issues) => errors.append(&mut issues),
213        }
214    }
215
216    if errors.is_empty() {
217        Ok(outputs)
218    } else {
219        Err(errors)
220    }
221}
