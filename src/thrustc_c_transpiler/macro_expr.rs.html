20use crate::macro_ast::{MacroExpr, MacroPostOp, MacroUnOp};
21
22impl MacroExpr {
23    #[inline]
24    pub fn is_atomic(&self) -> bool {
25        matches!(
26            self,
27            MacroExpr::Ident(_)
28                | MacroExpr::Literal(_)
29                | MacroExpr::Null
30                | MacroExpr::Paren(_)
31                | MacroExpr::Call { .. }
32                | MacroExpr::Index { .. }
33                | MacroExpr::Member { .. }
34                | MacroExpr::Postfix { .. }
35        )
36    }
37}
38
39impl MacroExpr {
40    #[inline]
41    pub fn base_identifier(&self) -> Option<&str> {
42        match self {
43            MacroExpr::Ident(name) => Some(name),
44            MacroExpr::Paren(inner) => Self::base_identifier(inner),
45            _ => None,
46        }
47    }
48}
49
50impl MacroExpr {
51    /// Returns `(is_pointer, pointee_is_pointer)` for an operand identifier.
52    #[inline]
53    fn pointer_info(&self, ctx: &crate::macros::MacroContext<'_>) -> (bool, bool) {
54        let ty: Option<&crate::macro_type::InferredType> = self
55            .base_identifier()
56            .and_then(|name| ctx.get_type_environment().get(name));
57
58        match ty {
59            Some(crate::macro_type::InferredType::Pointer(inner)) => (
60                true,
61                matches!(inner.as_ref(), crate::macro_type::InferredType::Pointer(_)),
62            ),
63            _ => (false, false),
64        }
65    }
66}

68pub fn lower(
69    ctx: &mut crate::macros::MacroContext,
70    expr: &MacroExpr,
71    location: crate::location::Location,
72    result_type: &str,
73) -> String {
74    match expr {
75        MacroExpr::Ident(name) => name.to_string(),
76        MacroExpr::Literal(text) => text.to_string(),
77        MacroExpr::Null => "nullptr".to_string(),
78
79        MacroExpr::Paren(inner) => {
80            let inner_text: String = self::lower(ctx, inner, location, result_type);
81
82            format!("({inner_text})")
83        }
84
85        MacroExpr::Unary { op, arg } => {
86            if matches!(op, MacroUnOp::Ref) {
87                let arg_text: String =
88                    self::lower(ctx, arg, crate::location::Location::AddressOf, result_type);
89
90                let grouped: String = if arg.is_atomic() {
91                    arg_text
92                } else {
93                    format!("({arg_text})")
94                };
95
96                return format!("ref ({grouped})");
97            }
98
99            let arg_text: String =
100                self::lower(ctx, arg, crate::location::Location::RValue, result_type);
101
102            let grouped: String = if arg.is_atomic() {
103                arg_text.clone()
104            } else {
105                format!("({arg_text})")
106            };
107
108            if matches!(op, MacroUnOp::PreIncrement | MacroUnOp::PreDecrement) {
109                let (is_pointer, pointee_is_pointer) = arg.pointer_info(ctx);
110
111                if is_pointer {
112                    return crate::pointer::Pointer::lower_advance(
113                        &arg_text,
114                        "1",
115                        matches!(op, MacroUnOp::PreIncrement),
116                        pointee_is_pointer,
117                    );
118                }
119            }
120
121            match op {
122                MacroUnOp::Ref => format!("ref ({grouped})"),
123                MacroUnOp::Deref => format!("deref {grouped}"),
124                MacroUnOp::Not => format!("!{grouped}"),
125                MacroUnOp::Invert => format!("~{grouped}"),
126                MacroUnOp::Negate => format!("-{grouped}"),
127                MacroUnOp::Positive => format!("+{grouped}"),
128                MacroUnOp::PreIncrement => format!("++{grouped}"),
129                MacroUnOp::PreDecrement => format!("--{grouped}"),
130            }
131        }
132
133        MacroExpr::Postfix { op, arg } => {
134            let arg_text: String =
135                self::lower(ctx, arg, crate::location::Location::RValue, result_type);
136
137            let grouped: String = if arg.is_atomic() {
138                arg_text.clone()
139            } else {
140                format!("({arg_text})")
141            };
142
143            let (is_pointer, pointee_is_pointer) = arg.pointer_info(ctx);
144
145            if is_pointer {
146                return crate::pointer::Pointer::lower_advance(
147                    &arg_text,
148                    "1",
149                    matches!(op, MacroPostOp::Increment),
150                    pointee_is_pointer,
151                );
152            }
153
154            match op {
155                MacroPostOp::Increment => format!("{grouped}++"),
156                MacroPostOp::Decrement => format!("{grouped}--"),
157            }
158        }
159
160        MacroExpr::Binary { op, left, right } => {
161            let left_text: String =
162                self::lower(ctx, left, crate::location::Location::RValue, result_type);
163            let right_text: String =
164                self::lower(ctx, right, crate::location::Location::RValue, result_type);
165
166            let (left_is_pointer, left_pointee_is_pointer) = left.pointer_info(ctx);
167            let (right_is_pointer, right_pointee_is_pointer) = right.pointer_info(ctx);
168
169            let pointee_is_pointer: bool = if left_is_pointer {
170                left_pointee_is_pointer
171            } else {
172                right_pointee_is_pointer
173            };
174
175            if let Some(transformed) = crate::pointer::Pointer::lower_binary(
176                left_is_pointer,
177                right_is_pointer,
178                pointee_is_pointer,
179                op,
180                &left_text,
181                &right_text,
182            ) {
183                return transformed;
184            }
185
186            format!("{left_text} {op} {right_text}")
187        }
188
189        MacroExpr::Comma(items) => {
190            let item_texts: Vec<String> = items
191                .iter()
192                .map(|item| self::lower(ctx, item, crate::location::Location::RValue, result_type))
193                .collect();
194
195            format!("({})", item_texts.join(", "))
196        }
197
198        MacroExpr::Assign { op, target, value } => {
199            let target_text: String =
200                self::lower(ctx, target, crate::location::Location::LValue, result_type);
201            let value_text: String =
202                self::lower(ctx, value, crate::location::Location::RValue, result_type);
203
204            let (is_pointer, pointee_is_pointer) = target.pointer_info(ctx);
205
206            if is_pointer && (op == "+=" || op == "-=") {
207                return crate::pointer::Pointer::lower_advance(
208                    &target_text,
209                    &value_text,
210                    op == "+=",
211                    pointee_is_pointer,
212                );
213            }
214
215            format!("{target_text} {op} {value_text}")
216        }
217
218        MacroExpr::Ternary {
219            cond,
220            then_branch,
221            else_branch,
222        } => {
223            let cond_text: String = if let MacroExpr::Binary { op, .. } = cond.as_ref() {
224                if op == "=="
225                    || op == "!="
226                    || op == "<"
227                    || op == "<="
228                    || op == ">"
229                    || op == ">="
230                    || op == "&&"
231                    || op == "||"
232                {
233                    self::lower(ctx, cond, crate::location::Location::RValue, "bool")
234                } else {
235                    format!(
236                        "({}) != 0",
237                        self::lower(ctx, cond, crate::location::Location::RValue, "bool")
238                    )
239                }
240            } else {
241                format!(
242                    "({}) != 0",
243                    self::lower(ctx, cond, crate::location::Location::RValue, "bool")
244                )
245            };
246
247            let then_type: Option<crate::macro_type::InferredType> =
248                crate::macro_type::MacroType::expression_type(ctx, then_branch);
249            let else_type: Option<crate::macro_type::InferredType> =
250                crate::macro_type::MacroType::expression_type(ctx, else_branch);
251
252            let ternary_type: String = then_type
253                .or(else_type)
254                .map(|ty| ty.text())
255                .unwrap_or_else(|| result_type.to_string());
256
257            let then_text: String = self::lower(ctx, then_branch, location, &ternary_type);
258            let else_text: String = self::lower(ctx, else_branch, location, &ternary_type);
259
260            let temporary: String = ctx.next_temporary_name();
261
262            ctx.push_pending_statement(format!("var {temporary}: {ternary_type};"));
263            ctx.push_pending_statement(format!("if {cond_text} {{"));
264            ctx.push_pending_statement(format!(
265                "    {temporary} = ({then_text}) as {ternary_type};"
266            ));
267            ctx.push_pending_statement("} else {".into());
268            ctx.push_pending_statement(format!(
269                "    {temporary} = ({else_text}) as {ternary_type};"
270            ));
271            ctx.push_pending_statement("}".into());
272
273            temporary
274        }
275
276        MacroExpr::Call { callee, args } => {
277            let callee_text: String =
278                self::lower(ctx, callee, crate::location::Location::RValue, result_type);
279
280            let grouped_callee: String = if matches!(callee.as_ref(), MacroExpr::Ident(_)) {
281                callee_text
282            } else {
283                format!("({callee_text})")
284            };
285
286            let argument_texts: Vec<String> = args
287                .iter()
288                .map(|arg| self::lower(ctx, arg, crate::location::Location::RValue, result_type))
289                .collect();
290
291            format!("{grouped_callee}({})", argument_texts.join(", "))
292        }
293
294        MacroExpr::Index { base, index } => {
295            fn lower_base(
296                ctx: &mut crate::macros::MacroContext,
297                expr: &MacroExpr,
298                result_type: &str,
299            ) -> String {
300                match expr {
301                    MacroExpr::Ident(name) => name.to_string(),
302
303                    MacroExpr::Paren(inner) => lower_base(ctx, inner, result_type),
304
305                    MacroExpr::Member { base, field } => {
306                        let base_text: String = lower_base(ctx, base, result_type);
307
308                        format!("{base_text}.{field}")
309                    }
310
311                    _ => format!(
312                        "({})",
313                        self::lower(ctx, expr, crate::location::Location::RValue, result_type)
314                    ),
315                }
316            }
317
318            let index_text: String =
319                self::lower(ctx, index, crate::location::Location::RValue, result_type);
320
321            if location.is_address_of() {
322                let base_text: String = self::lower(ctx, base, location, result_type);
323
324                return format!("{base_text}[{index_text}]");
325            }
326
327            let base_text: String = lower_base(ctx, base, result_type);
328
329            format!("{base_text}->[{index_text}]")
330        }
331
332        MacroExpr::Member { base, field } => {
333            let base_text: String =
334                self::lower(ctx, base, crate::location::Location::RValue, result_type);
335
336            let grouped_base: String = if matches!(base.as_ref(), MacroExpr::Ident(_)) {
337                base_text
338            } else {
339                format!("({base_text})")
340            };
341
342            if location.is_address_of() {
343                format!("{grouped_base}.{field}")
344            } else {
345                format!("{grouped_base}->{field}")
346            }
347        }
348
349        MacroExpr::Cast { target, arg } => {
350            let arg_text: String =
351                self::lower(ctx, arg, crate::location::Location::RValue, result_type);
352
353            format!("({arg_text}) as {target}")
354        }
355
356        MacroExpr::SizeOf(target) => {
357            format!("abiSizeOf({target})")
358        }
359    }
360}

362pub fn lower_function_body(
363    ctx: &mut crate::macros::MacroContext,
364    expr: &MacroExpr,
365    return_type: &str,
366) -> String {
367    let body_text: String = self::lower(ctx, expr, crate::location::Location::RValue, return_type);
368
369    format!("({body_text}) as {return_type}")
370}
