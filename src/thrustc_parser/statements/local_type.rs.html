20use thrustc_ast::{Ast, NodeId};
21use thrustc_attributes::ThrustAttributes;
22use thrustc_errors::{CompilationIssue, CompilationIssueCode};
23use thrustc_code_location::Span;
24use thrustc_token::{Token, traits::TokenExtensions};
25use thrustc_token_type::TokenType;
26use thrustc_typesystem::Type;
27
28use crate::{ParserContext, attributes, typegeneration};
29
30pub fn parse_custom_type_stmt<'parser>(
31    ctx: &mut ParserContext<'parser>,
32) -> Result<Ast<'parser>, CompilationIssue> {
33    ctx.consume(
34        TokenType::Type,
35        CompilationIssueCode::E0001,
36        "Expected 'type' keyword.".into(),
37    )?;
38
39    let name_tk: &Token = ctx.consume(
40        TokenType::Identifier,
41        CompilationIssueCode::E0001,
42        "Expected identifier.".into(),
43    )?;
44
45    let name: &str = name_tk.get_lexeme();
46    let span: Span = name_tk.get_span();
47
48    ctx.consume(
49        TokenType::Eq,
50        CompilationIssueCode::E0001,
51        String::from("Expected '='."),
52    )?;
53
54    let attributes: ThrustAttributes =
55        attributes::build_compiler_attributes(ctx, &[TokenType::LBrace])?;
56
57    let custom_type: Type = typegeneration::build_type(ctx, false)?;
58
59    ctx.consume(
60        TokenType::SemiColon,
61        CompilationIssueCode::E0001,
62        "Expected ';'.".into(),
63    )?;
64
65    if !ctx.is_main_scope() {
66        ctx.get_mut_symbols()
67            .new_custom_type(name, (custom_type.clone(), attributes), span)?;
68
69        Ok(Ast::CustomType {
70            name: name.to_string(),
71            kind: custom_type,
72            span,
73            id: NodeId::new(),
74        })
75    } else {
76        Ok(Ast::invalid_ast(span))
77    }
78}
