20use crate::{
21    Type,
22    traits::{ConstantTypeExtensions, TypeFixedArrayEntensions},
23};
24
25impl TypeFixedArrayEntensions for Type {
26    #[inline]
27    fn get_fixed_array_base_type(&self) -> Type {
28        let non_constant_type: Type = self.remove_all_constant_type();
29
30        if let Type::FixedArray { base_type, .. } = non_constant_type {
31            return (*base_type).clone();
32        }
33
34        if let Type::Ptr {
35            subtype: Some(inner),
36            ..
37        } = non_constant_type
38        {
39            return inner.get_fixed_array_base_type();
40        }
41
42        if let Type::Const(inner, ..) = non_constant_type {
43            return inner.get_fixed_array_base_type();
44        }
45
46        self.clone()
47    }
48
49    #[inline]
50    fn get_fixed_array_type_herarchy(&self) -> u8 {
51        match self {
52            Type::Bool { .. } => 1,
53            Type::Char { .. } => 2,
54
55            Type::U8 { .. } => 3,
56            Type::U16 { .. } => 4,
57            Type::U32 { .. } => 5,
58            Type::U64 { .. } => 6,
59            Type::U128 { .. } => 7,
60            Type::USize { .. } => 8,
61
62            Type::S8 { .. } => 9,
63            Type::S16 { .. } => 10,
64            Type::S32 { .. } => 11,
65            Type::S64 { .. } => 12,
66            Type::SSize { .. } => 13,
67
68            Type::F32 { .. } => 15,
69            Type::F64 { .. } => 16,
70            Type::F128 { .. } => 17,
71            Type::FX8680 { .. } => 18,
72            Type::FPPC128 { .. } => 19,
73
74            Type::Const(subtype, ..) => subtype.get_fixed_array_type_herarchy(),
75
76            Type::Ptr {
77                subtype: Some(subtype),
78                ..
79            } => subtype.get_fixed_array_type_herarchy(),
80            Type::Ptr { subtype: None, .. } => 20,
81
82            Type::Array { .. } => 21,
83            Type::FixedArray { .. } => 22,
84            Type::Struct { .. } => 23,
85
86            Type::Fn { .. } => 24,
87            Type::Void { .. } => 25,
88            Type::Unresolved { .. } => 26,
89        }
90    }
91}
