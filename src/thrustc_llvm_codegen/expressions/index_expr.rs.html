20use inkwell::values::{BasicValueEnum, IntValue, PointerValue};
21use thrustc_ast::{Ast, ast_metadata::IndexMetadata, traits::AstCodeLocation};
22use thrustc_code_location::Span;
23use thrustc_typesystem::{
24    Type,
25    traits::{
26        ConstantTypeExtensions, IndexExtensions, InfererTypeExtensions, TypeIsExtensions,
27        TypePointerExtensions,
28    },
29};
30
31use crate::{
32    codegen,
33    context::{CodeGenLocation, LLVMCodeGenContext},
34    expressions, memory,
35    traits::AstLLVMGetType,
36};
37
38pub fn compile<'ctx>(
39    context: &mut LLVMCodeGenContext<'_, 'ctx>,
40    source: &'ctx Ast<'ctx>,
41    index: &'ctx Ast<'ctx>,
42    metadata: &IndexMetadata,
43) -> BasicValueEnum<'ctx> {
44    let source_type: Type = source.get_type_for_llvm().remove_all_constant_type();
45
46    if source_type.is_native_vector_type() {
47        let span: Span = index.get_span();
48        let index_type: Type = Type::U32 { span };
49
50        context.add_codegen_location(CodeGenLocation::RValue);
51
52        let vector: BasicValueEnum = codegen::compile_as_value(context, source, None);
53        let index: IntValue =
54            codegen::compile_as_value(context, index, Some(&index_type)).into_int_value();
55
56        context.pop_current_codegen_location();
57
58        context
59            .get_llvm_builder()
60            .build_extract_element(vector.into_vector_value(), index, "")
61            .unwrap_or_else(|_| {
62                crate::abort::abort_codegen(
63                    context,
64                    "Failed to extract native vector element.",
65                    span,
66                    std::path::PathBuf::from(file!()),
67                    line!(),
68                )
69            })
70    } else {
71        context.add_codegen_location(CodeGenLocation::RValue);
72        let source_value: BasicValueEnum<'_> = codegen::compile_as_ptr_value(context, source, None);
73        context.pop_current_codegen_location();
74
75        let ptr_value: PointerValue<'_> = source_value.into_pointer_value();
76        let mut ptr_type: &Type = &source_type;
77        let infered_inner_type: Type = ptr_type.get_inferer_inner_type();
78
79        let ordered_indexes: Vec<IntValue> = {
80            let span: Span = index.get_span();
81
82            let has_inferer_inner_type: bool = ptr_type.has_infered_inner_type();
83
84            if has_inferer_inner_type {
85                ptr_type = &infered_inner_type;
86            }
87
88            let is_ptr_aggv_type: bool = ptr_type.is_ptr_fixed_array_type();
89            let is_ptr_like_type: bool = ptr_type.is_ptr_like_type();
90
91            context.add_codegen_location(CodeGenLocation::RValue);
92
93            let indexes: Vec<IntValue> = if is_ptr_aggv_type {
94                let base_type: Type = Type::U32 { span };
95
96                let base: IntValue = expressions::literal_integer_expr::compile(
97                    context,
98                    &base_type,
99                    0,
100                    index.get_span(),
101                );
102
103                let depth_type: Type = Type::U32 { span };
104
105                let depth: IntValue =
106                    codegen::compile_as_value(context, index, Some(&depth_type)).into_int_value();
107
108                vec![base, depth]
109            } else if is_ptr_like_type {
110                let base_type: Type = Type::U64 { span };
111
112                let base: IntValue =
113                    codegen::compile_as_value(context, index, Some(&base_type)).into_int_value();
114
115                vec![base]
116            } else {
117                let base_type: Type = Type::U32 { span };
118
119                let base: IntValue = expressions::literal_integer_expr::compile(
120                    context,
121                    &base_type,
122                    0,
123                    index.get_span(),
124                );
125
126                let depth_type: Type = Type::U32 { span };
127
128                let depth: IntValue =
129                    codegen::compile_as_value(context, index, Some(&depth_type)).into_int_value();
130
131                vec![base, depth]
132            };
133
134            context.pop_current_codegen_location();
135
136            indexes
137        };
138
139        let span: Span = source.get_span();
140
141        let ptr: PointerValue<'_> =
142            memory::gep_anon(context, ptr_value, ptr_type, &ordered_indexes, span);
143
144        if metadata.is_deref() {
145            if context.get_codegen_location().is_direct_behavior() {
146                return ptr.into();
147            }
148
149            let element_type: Type = ptr_type.calculate_index_type(1).clone();
150
151            return memory::dereference(context, ptr, &element_type, span);
152        }
153
154        if context.get_codegen_location().is_load_behavior() {
155            let element_type: Type = ptr_type.calculate_index_type(1).clone();
156
157            if element_type.is_ptr_like_type()
158                && !element_type.is_struct_type()
159                && !element_type.is_fixed_array_type()
160                && !element_type.is_array_type_with_inference()
161            {
162                return memory::load_pointer(context, ptr, span);
163            } else {
164                return ptr.into();
165            }
166        }
167
168        ptr.into()
169    }
170}
