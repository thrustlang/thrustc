38pub fn compile<'ctx>(
39    context: &mut LLVMCodeGenContext<'_, 'ctx>,
40    source: &'ctx Ast<'ctx>,
41    data: &'ctx PropertyData,
42) -> BasicValueEnum<'ctx> {
43    let source_type: &Type = source.get_type_for_llvm();
44
45    let is_allocated: bool = source.is_memory_assigned_value().unwrap_or_else(|_| {
46        abort::abort_codegen(
47            context,
48            "Failed to compile the property!",
49            source.get_span(),
50            std::path::PathBuf::from(file!()),
51            line!(),
52        )
53    });
54
55    if (is_allocated && source_type.is_struct_type()) || source_type.is_ptr_composite_type() {
56        self::compile_gep_property(context, source, data)
57    } else {
58        self::compile_extract_property(context, source, data)
59    }
60}

62fn compile_extract_property<'ctx>(
63    context: &mut LLVMCodeGenContext<'_, 'ctx>,
64    source: &'ctx Ast<'ctx>,
65    data: &'ctx PropertyData,
66) -> BasicValueEnum<'ctx> {
67    let llvm_builder: &Builder = context.get_llvm_builder();
68
69    let span: Span = source.get_span();
70
71    let mut property: BasicValueEnum = {
72        let value: BasicValueEnum = codegen::compile_as_value(context, source, None);
73
74        let index: u32 = data
75            .get_first_property()
76            .unwrap_or_else(|| {
77                abort::abort_codegen(
78                    context,
79                    "Failed to compile the property!",
80                    span,
81                    std::path::PathBuf::from(file!()),
82                    line!(),
83                )
84            })
85            .get_index();
86
87        llvm_builder
88            .build_extract_value(value.into_struct_value(), index, "")
89            .unwrap_or_else(|_| {
90                abort::abort_codegen(
91                    context,
92                    "Failed to compile the property!",
93                    span,
94                    std::path::PathBuf::from(file!()),
95                    line!(),
96                )
97            })
98    };
99
100    for field in data.iter().skip(1) {
101        let index: u32 = field.get_index();
102
103        property = llvm_builder
104            .build_extract_value(property.into_struct_value(), index, "")
105            .unwrap_or_else(|_| {
106                abort::abort_codegen(
107                    context,
108                    "Failed to compile the property!",
109                    span,
110                    std::path::PathBuf::from(file!()),
111                    line!(),
112                )
113            });
114    }
115
116    property
117}

119fn compile_gep_property<'ctx>(
120    context: &mut LLVMCodeGenContext<'_, 'ctx>,
121    source: &'ctx Ast<'ctx>,
122    data: &'ctx PropertyData,
123) -> BasicValueEnum<'ctx> {
124    let llvm_builder: &Builder = context.get_llvm_builder();
125
126    let span: Span = source.get_span();
127
128    let source_value: BasicValueEnum<'_> = codegen::compile_as_ptr_value(context, source, None);
129
130    let ptr_value: PointerValue = source_value.into_pointer_value();
131    let ptr_type: &Type = source.get_type_for_llvm();
132
133    let index: u32 = data
134        .get_first_property()
135        .unwrap_or_else(|| {
136            abort::abort_codegen(
137                context,
138                "Failed to compile the property!",
139                span,
140                std::path::PathBuf::from(file!()),
141                line!(),
142            )
143        })
144        .get_index();
145
146    let mut property_value: PointerValue =
147        memory::gep_struct_anon(context, ptr_value, ptr_type, index, span);
148
149    for field in data.iter().skip(1) {
150        let base_type: Type = field.get_base_type();
151        let index: u32 = field.get_index();
152
153        let property_type: BasicTypeEnum =
154            typegeneration::generate_pointer_arithmetic_type(context, &base_type);
155
156        property_value = llvm_builder
157            .build_struct_gep(property_type, property_value, index, "")
158            .unwrap_or_else(|_| {
159                abort::abort_codegen(
160                    context,
161                    "Failed to gep a value from pointer!",
162                    span,
163                    std::path::PathBuf::from(file!()),
164                    line!(),
165                )
166            });
167    }
168
169    property_value.into()
170}
