20use thrustc_ast::Ast;
21use thrustc_ast::traits::AstCodeLocation;
22use thrustc_span::Span;
23
24use crate::abort;
25use crate::block;
26use crate::codegen::LLVMCodegen;
27use crate::traits::LLVMFunctionExtensions;
28
29use inkwell::basic_block::BasicBlock;
30use inkwell::builder::Builder;
31use inkwell::values::FunctionValue;
32
33pub fn compile<'ctx>(codegen: &mut LLVMCodegen<'_, 'ctx>, node: &'ctx Ast<'ctx>) {
34    let llvm_builder: &Builder = codegen.get_context().get_llvm_builder();
35
36    let llvm_function: FunctionValue = codegen
37        .get_mut_context()
38        .get_current_function(node.get_span())
39        .get_value();
40
41    let Ast::Loop { block, span, .. } = node else {
42        return;
43    };
44
45    let block_span: Span = block.get_span();
46
47    let start: BasicBlock = block::append_block(codegen.get_context(), llvm_function);
48    let exit: BasicBlock = block::append_block(codegen.get_context(), llvm_function);
49
50    llvm_builder
51        .build_unconditional_branch(start)
52        .unwrap_or_else(|_| {
53            abort::abort_codegen(
54                codegen.get_mut_context(),
55                "Failed to compile loop terminator to start!",
56                *span,
57                std::path::PathBuf::from(file!()),
58                line!(),
59            )
60        });
61
62    llvm_builder.position_at_end(start);
63
64    if codegen.get_context().get_loop_ctx().get_all_branch_depth() == 0 {
65        codegen
66            .get_mut_context()
67            .get_mut_loop_context()
68            .set_breakall_branch(exit);
69
70        codegen
71            .get_mut_context()
72            .get_mut_loop_context()
73            .set_continueall_branch(start);
74    }
75
76    codegen
77        .get_mut_context()
78        .get_mut_loop_context()
79        .add_continue_branch(start);
80
81    codegen
82        .get_mut_context()
83        .get_mut_loop_context()
84        .add_break_branch(exit);
85
86    codegen.codegen_block(block);
87
88    if codegen
89        .get_mut_context()
90        .get_last_builder_block(block_span)
91        .get_terminator()
92        .is_none()
93    {
94        llvm_builder
95            .build_unconditional_branch(start)
96            .unwrap_or_else(|_| {
97                abort::abort_codegen(
98                    codegen.get_mut_context(),
99                    "Failed to compile loop body terminator to start!",
100                    *span,
101                    std::path::PathBuf::from(file!()),
102                    line!(),
103                )
104            });
105    }
106
107    codegen.get_mut_context().get_mut_loop_context().pop();
108
109    if codegen.get_context().get_loop_ctx().get_branch_depth() == 0 {
110        codegen
111            .get_mut_context()
112            .get_mut_loop_context()
113            .pop_superior_branch();
114    }
115
116    llvm_builder.position_at_end(exit);
117}
