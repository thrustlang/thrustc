32#[inline]
33#[must_use]
34pub fn generate_string(
35    context: &LLVMCodeGenContext<'_, '_>,
36    range: std::ops::RangeInclusive<usize>,
37) -> String {
38    if !context.get_compiler_options().need_obfuscate_ir() {
39        String::new()
40    } else {
41        let length: usize = fastrand::usize(range);
42        let mut random_string: String = String::with_capacity(length);
43
44        for _ in 0..length {
45            let n: u8 = fastrand::u8(0..52);
46
47            let c: char = match n {
48                0..=25 => (b'A' + n) as char,
49                26..=51 => (b'a' + (n - 26)) as char,
50                _ => '_',
51            };
52
53            random_string.push(c);
54        }
55
56        random_string
57    }
58}

60pub fn clean_llvm_name(name: &std::ffi::CStr) -> std::borrow::Cow<'_, str> {
61    let s: std::borrow::Cow<'_, str> = name.to_string_lossy();
62
63    if let Some(dot_pos) = s.rfind('.') {
64        let suffix = &s[dot_pos + 1..];
65        if !suffix.is_empty() && suffix.chars().all(|c| c.is_ascii_digit()) {
66            return std::borrow::Cow::Owned(s[..dot_pos].to_string());
67        }
68    }
69
70    s
71}

73pub fn get_functions_by_ordered_calls<'ctx>(
74    functions: Vec<FunctionValue<'ctx>>,
75) -> Vec<FunctionValue<'ctx>> {
76    let mut ordered: Vec<FunctionValue> = Vec::with_capacity(functions.len());
77    let mut visited: HashSet<CString> = HashSet::with_capacity(functions.len());
78
79    let allowed: HashSet<CString> = functions.iter().map(|f| f.get_name().to_owned()).collect();
80
81    for function in &functions {
82        dfs_post_order(function, &allowed, &mut visited, &mut ordered);
83    }
84
85    ordered
86}
87
88fn dfs_post_order<'ctx>(
89    function: &FunctionValue<'ctx>,
90    allowed: &HashSet<CString>,
91    visited: &mut HashSet<CString>,
92    ordered: &mut Vec<FunctionValue<'ctx>>,
93) {
94    let name: CString = function.get_name().to_owned();
95
96    if visited.contains(&name) {
97        return;
98    }
99
100    visited.insert(name.clone());
101
102    for basic_block in function.get_basic_blocks() {
103        let mut instr: Option<InstructionValue<'_>> = basic_block.get_first_instruction();
104
105        while let Some(current_instr) = instr {
106            if current_instr.get_opcode() == InstructionOpcode::Call {
107                let callsite: CallSiteValue<'_> =
108                    unsafe { CallSiteValue::new(current_instr.as_value_ref()) };
109                let called_fn: FunctionValue<'_> = callsite.get_called_fn_value();
110
111                let called_name: CString = called_fn.get_name().to_owned();
112
113                if allowed.contains(&called_name) {
114                    dfs_post_order(&called_fn, allowed, visited, ordered);
115                }
116            }
117
118            instr = current_instr.get_next_instruction();
119        }
120    }
121
122    ordered.push(*function);
123}
