51pub fn compile<'ctx>(context: &mut LLVMCodeGenContext<'_, 'ctx>, asm_fn: AssemblerFunction<'ctx>) {
52    let llvm_module: &Module = context.get_llvm_module();
53    let llvm_context: &Context = context.get_llvm_context();
54    let llvm_builder: &Builder = context.get_llvm_builder();
55
56    let name: &str = asm_fn.0;
57    let ascii_name: &str = &asm_fn.1.replace("\0", "");
58
59    let assembler: String = asm_fn.2.to_string();
60    let constraints: String = asm_fn.3.to_string();
61    let return_type: &Type = asm_fn.4;
62    let parameters: &[Ast] = asm_fn.5;
63    let parameters_types: &[Type] = asm_fn.6;
64    let attributes: LLVMAttributes = thrustc_llvm_attributes::into_llvm_attributes(asm_fn.7);
65
66    let span: Span = asm_fn.8;
67
68    let sideeffects: bool = attributes.has_asmsideffects_attribute();
69    let align_stack: bool = attributes.has_asmalignstack_attribute();
70    let can_throw: bool = attributes.has_asmthrow_attribute();
71    let is_public: bool = attributes.has_public_attribute();
72
73    let call_convention: u32 = if let Some(LLVMAttribute::Convention(call_convention, ..)) =
74        attributes.get_attr(LLVMAttributeComparator::Convention)
75    {
76        call_convention as u32
77    } else {
78        LLVMCallConvention::Standard as u32
79    };
80
81    let syntax: InlineAsmDialect = match attributes.get_attr(LLVMAttributeComparator::AsmSyntax) {
82        Some(LLVMAttribute::AsmSyntax(syntax), ..) => match syntax {
83            "Intel" => InlineAsmDialect::Intel,
84            "AT&T" => InlineAsmDialect::ATT,
85
86            _ => InlineAsmDialect::Intel,
87        },
88        _ => InlineAsmDialect::Intel,
89    };
90
91    let llvm_function_name: String = if is_public {
92        format!("__asm_fn_{}", ascii_name)
93    } else {
94        format!(
95            "__asm_fn_{}_{}",
96            utils::generate_string(context, utils::LONG_RANGE_OBFUSCATION),
97            ascii_name
98        )
99    };
100
101    let generated_function_type: (
102        FunctionType<'_>,
103        Option<thrustc_llvm_abi::LLVMABIConfiguration>,
104    ) = typegeneration::compile_as_function_type(
105        context,
106        return_type,
107        parameters,
108        false,
109        CompilerFunctionVariant::AssemblerFunction,
110    );
111
112    let function_type: FunctionType<'_> = generated_function_type.0;
113    let function_abi_config: Option<thrustc_llvm_abi::LLVMABIConfiguration> =
114        generated_function_type.1;
115
116    let function_ptr: PointerValue = llvm_context.create_inline_asm(
117        function_type,
118        assembler,
119        constraints,
120        sideeffects,
121        align_stack,
122        Some(syntax),
123        can_throw,
124    );
125
126    let asm_function: FunctionValue =
127        llvm_module.add_function(&llvm_function_name, function_type, None);
128
129    let applicant: LLVMAttributeApplicant<'_> = LLVMAttributeApplicant::AsmFunction {
130        value: asm_function,
131        span,
132    };
133
134    AttributeBuilder::add_function_attributes(context, &attributes, applicant);
135
136    let last_block: BasicBlock = context.get_last_builder_block(span);
137    let function_block: BasicBlock = block::append_block(context, asm_function);
138
139    llvm_builder.position_at_end(function_block);
140
141    let args: Vec<BasicMetadataValueEnum> = asm_function
142        .get_param_iter()
143        .map(|param| param.into())
144        .collect();
145
146    if let Ok(asm_fn_call) =
147        llvm_builder.build_indirect_call(function_type, function_ptr, &args, "")
148    {
149        match (
150            return_type.is_void_type(),
151            asm_fn_call.try_as_basic_value().left(),
152        ) {
153            (false, Some(return_value)) => {
154                llvm_builder
155                    .build_return(Some(&return_value))
156                    .map_err(|_| {
157                        abort::abort_codegen(
158                            context,
159                            "Failed to compile assembly function!",
160                            span,
161                            std::path::PathBuf::from(file!()),
162                            line!(),
163                        );
164                    })
165                    .ok();
166            }
167            _ => {
168                llvm_builder
169                    .build_return(None)
170                    .map_err(|_| {
171                        abort::abort_codegen(
172                            context,
173                            "Failed to compile assembly function!",
174                            span,
175                            std::path::PathBuf::from(file!()),
176                            line!(),
177                        );
178                    })
179                    .ok();
180            }
181        }
182    } else {
183        abort::abort_codegen(
184            context,
185            "Failed to compile indirect call for assembly function!",
186            span,
187            std::path::PathBuf::from(file!()),
188            line!(),
189        );
190    }
191
192    llvm_builder.position_at_end(last_block);
193
194    let prototype: LLVMFunction = (
195        asm_function,
196        return_type,
197        parameters_types,
198        call_convention,
199        function_abi_config,
200        attributes,
201        false,
202        span,
203    );
204
205    context.add_function(name, prototype);
206}
