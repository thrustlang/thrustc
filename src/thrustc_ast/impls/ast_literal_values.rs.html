20use crate::{Ast, traits::AstLiteralExtensions};
21
22impl AstLiteralExtensions for Ast<'_> {
23    fn is_totaly_literal_value(&self) -> bool {
24        match self {
25            Ast::Integer { .. }
26            | Ast::Float { .. }
27            | Ast::Boolean { .. }
28            | Ast::Char { .. }
29            | Ast::CString { .. }
30            | Ast::CNString { .. }
31            | Ast::NullPtr { .. } => true,
32
33            Ast::FixedArray { items, .. } => {
34                items.iter().all(|item| item.is_totaly_literal_value())
35            }
36            Ast::Array { items, .. } => items.iter().all(|item| item.is_totaly_literal_value()),
37
38            Ast::EnumValue { value, .. } => value.is_totaly_literal_value(),
39
40            Ast::Group { node, .. } => node.is_totaly_literal_value(),
41            Ast::BinaryOp { left, right, .. } => {
42                left.is_totaly_literal_value() && right.is_totaly_literal_value()
43            }
44            Ast::UnaryOp { node, .. } => node.is_totaly_literal_value(),
45
46            // No es un valor literal completo.
47            Ast::GlobalAssembler { .. }
48            | Ast::Index { .. }
49            | Ast::Embedded { .. }
50            | Ast::Struct { .. }
51            | Ast::Property { .. }
52            | Ast::Constructor { .. }
53            | Ast::If { .. }
54            | Ast::Elif { .. }
55            | Ast::Else { .. }
56            | Ast::For { .. }
57            | Ast::While { .. }
58            | Ast::Loop { .. }
59            | Ast::Continue { .. }
60            | Ast::Break { .. }
61            | Ast::ContinueAll { .. }
62            | Ast::BreakAll { .. }
63            | Ast::Block { .. }
64            | Ast::Defer { .. }
65            | Ast::CustomType { .. }
66            | Ast::Enum { .. }
67            | Ast::CompilerIntrinsic { .. }
68            | Ast::CompilerIntrinsicParameter { .. }
69            | Ast::AssemblerFunction { .. }
70            | Ast::AssemblerFunctionParameter { .. }
71            | Ast::Function { .. }
72            | Ast::FunctionParameter { .. }
73            | Ast::Return { .. }
74            | Ast::Static { .. }
75            | Ast::Const { .. }
76            | Ast::Var { .. }
77            | Ast::Reference { .. }
78            | Ast::Mutation { .. }
79            | Ast::Address { .. }
80            | Ast::Write { .. }
81            | Ast::Load { .. }
82            | Ast::Deref { .. }
83            | Ast::As { .. }
84            | Ast::GetLocation { .. }
85            | Ast::ModuleExpression { .. }
86            | Ast::Call { .. }
87            | Ast::IndirectCall { .. }
88            | Ast::AsmValue { .. }
89            | Ast::Builtin { .. }
90            | Ast::Import { .. }
91            | Ast::ImportC { .. }
92            | Ast::Unreachable { .. }
93            | Ast::Invalid { .. } => false,
94        }
95    }
96
97    #[inline]
98    fn is_literal_value(&self) -> bool {
99        matches!(
100            self,
101            Ast::Integer { .. }
102                | Ast::Float { .. }
103                | Ast::Boolean { .. }
104                | Ast::Char { .. }
105                | Ast::CString { .. }
106                | Ast::CNString { .. }
107                | Ast::NullPtr { .. }
108                | Ast::Array { .. }
109                | Ast::FixedArray { .. }
110        )
111    }
112
113    #[inline]
114    fn is_literal_ptr_value(&self) -> bool {
115        matches!(self, Ast::CString { .. } | Ast::CNString { .. })
116    }
117}
