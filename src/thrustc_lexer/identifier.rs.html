207pub fn lex(lexer: &mut Lexer) -> Result<(), CompilationIssue> {
208    while lexer.is_identifier_boundary(lexer.peek()) {
209        lexer.advance_only();
210    }
211
212    let lexem: String = lexer.lexeme();
213
214    if let Some(keyword) = KEYWORDS.get(lexem.as_str()) {
215        lexer.make(*keyword);
216    } else if let Some(atomic_stuff) = ATOMIC.get(lexem.as_str()) {
217        lexer.make(*atomic_stuff);
218    } else if let Some(attribute) = ATTRIBUTES.get(lexem.as_str()) {
219        lexer.make(*attribute);
220    } else if let Some(builtin) = BUILTINS.get(lexem.as_str()) {
221        lexer.make(*builtin);
222    } else if let Some(r#type) = TYPES.get(lexem.as_str()) {
223        lexer.make(*r#type);
224    } else {
225        lexer.make(TokenType::Identifier);
226    }
227
228    Ok(())
229}
