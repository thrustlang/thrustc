28#[derive(Debug, Clone)]
29pub enum CompilationIssue {
30    Error(CompilationIssueCode, String, String, Option<String>, Span),
31    Warning(CompilationIssueCode, String, Span),
32
33    FrontendBug(
34        String,
35        String,
36        Span,
37        CompilationPosition,
38        std::path::PathBuf,
39        u32,
40    ),
41
42    BackendBug(
43        String,
44        String,
45        Span,
46        CompilationPosition,
47        std::path::PathBuf,
48        u32,
49    ),
50}
51
52impl CompilationIssue {
53    #[inline]
54    pub fn is_bug(&self) -> bool {
55        matches!(
56            self,
57            CompilationIssue::FrontendBug(..) | CompilationIssue::BackendBug(..)
58        )
59    }
60}
